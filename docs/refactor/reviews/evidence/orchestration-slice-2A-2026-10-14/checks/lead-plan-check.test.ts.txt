/**
 * Lead acceptance check for slice 2A L1 (D3): EXPLAIN QUERY PLAN of every
 * ordered lookup shape. Not part of the suite; copied in, run once, removed.
 */
import { afterAll, expect, it } from 'vitest';
import { mkdtemp, rm } from 'node:fs/promises';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { DatabaseSync } from 'node:sqlite';
import { canonicalJson } from '../src/contracts/fingerprint.js';
import { createSqliteRecordBackend } from '../src/core/record-store/sqlite-record-store.js';
import type { RecordBackendSchemas } from '../src/core/record-store/ports.js';
import type { RecordOrderedLookupPort } from '../src/core/record-store/lookup-ports.js';
import { S2A_REQUEUE_LOOKUP_INDEXES, S2A_REQUEUE_BY_PROJECT_LOOKUP } from '../src/core/work-graph/tasks/claim-record-codecs.js';
import {
  S2A_ROUND_REVERIFY_LOOKUP_INDEXES, S2A_ROUND_REVERIFY_LOOKUP,
} from '../src/core/work-graph/persistence/evidence-record-codecs.js';

const captured: { db: DatabaseSync; sql: string }[] = [];
const originalPrepare = DatabaseSync.prototype.prepare;
DatabaseSync.prototype.prepare = function (this: DatabaseSync, sql: string) {
  if (sql.includes('ORDER BY json_extract')) captured.push({ db: this, sql });
  return originalPrepare.call(this, sql);
} as typeof DatabaseSync.prototype.prepare;

const pass = (aggregateType: string, schemaId: string) => ({
  schemaId, aggregateType, validate: (record: never) => ({ status: 'decoded' as const, value: record }),
});
const integerIndex = { name: 'lead-integer-by-group', aggregateType: 'Widget', paths: ['group'],
  orderBy: { path: 'rank', valueType: 'integer' as const } };
const schemas = {
  records: [pass('TaskRequeue', 'TaskRequeueSnapshot@1'), pass('VerificationRound', 'VerificationRoundSnapshot@1'),
    pass('Widget', 'Widget@1')],
  events: [],
  lookups: [...S2A_REQUEUE_LOOKUP_INDEXES, ...S2A_ROUND_REVERIFY_LOOKUP_INDEXES, integerIndex],
} as unknown as RecordBackendSchemas;

const dirs: string[] = [];
afterAll(async () => {
  DatabaseSync.prototype.prepare = originalPrepare;
  for (const dir of dirs) await rm(dir, { recursive: true, force: true });
});

it('every ordered lookup shape seeks its index and never sorts in a temp b-tree', async () => {
  const dir = await mkdtemp(join(tmpdir(), 'lead-plan-'));
  dirs.push(dir);
  const backend = createSqliteRecordBackend({ path: join(dir, 'plan.sqlite'), schemas });
  const records = backend.records as unknown as RecordOrderedLookupPort;
  const requeueKey = canonicalJson({ aggregateType: 'TaskRequeue', projectId: 'P', goalId: 'G', taskId: 'T', requeueId: 'R' });
  const roundKey = canonicalJson({ aggregateType: 'VerificationRound', projectId: 'P', roundId: 'R' });
  const widgetKey = canonicalJson({ aggregateType: 'Widget', projectId: 'P', widgetId: 'W' });
  const shapes = [
    { index: S2A_REQUEUE_BY_PROJECT_LOOKUP, values: ['P'], order: 'c0000000001', refKey: requeueKey },
    { index: S2A_ROUND_REVERIFY_LOOKUP, values: ['P', 'W', 'system'], order: 'c0000000001', refKey: roundKey },
    { index: integerIndex.name, values: ['g'], order: 7, refKey: widgetKey },
  ];
  const report: string[] = [];
  for (const shape of shapes) {
    for (const direction of ['asc', 'desc'] as const) {
      for (const variant of ['plain', 'after', 'bound+after'] as const) {
        const before = captured.length;
        const result = await records.lookupOrdered({
          index: shape.index, values: shape.values, direction, limit: 200,
          ...(variant === 'bound+after' ? { bound: { value: shape.order, inclusive: true } } : {}),
          ...(variant !== 'plain' ? { after: { order: shape.order, refKey: shape.refKey } } : {}),
        } as never);
        report.push(`${shape.index} ${direction} ${variant}: ${JSON.stringify(result.status)}`);
        const sql = captured.at(-1);
        if (captured.length === before) console.log(`NO SQL for ${report.at(-1)}: ${JSON.stringify(result)}`);
        expect(captured.length).toBeGreaterThan(before);
        if (sql === undefined) throw new Error('no SQL captured');
        const plan = originalPrepare.call(sql.db, `EXPLAIN QUERY PLAN ${sql.sql}`).all() as { detail: string }[];
        const details = plan.map(row => row.detail);
        report.push(`  SQL: ${sql.sql}`);
        for (const detail of details) report.push(`  PLAN: ${detail}`);
        console.log(report.splice(0).join('\n'));
        expect(details.some(detail => detail.includes('TEMP B-TREE'))).toBe(false);
        // Which index the planner picks is reported; the gate is a SEARCH (seek) with no sort.
        expect(details.some(detail => /^SEARCH snapshots USING (COVERING )?INDEX next_lookup_/.test(detail))).toBe(true);
      }
    }
  }
  console.log(report.join('\n'));
  await backend.close();
});

it('a deep continuation page costs about the same as the first page (the continuation seeks)', async () => {
  const { StatementSync } = await import('node:sqlite');
  const calls: { db: DatabaseSync; sql: string; args: unknown[] }[] = [];
  const originalAll = StatementSync.prototype.all;
  const dbOf = new WeakMap<object, DatabaseSync>();
  DatabaseSync.prototype.prepare = function (this: DatabaseSync, sql: string) {
    const statement = originalPrepare.call(this, sql);
    dbOf.set(statement, this);
    return statement;
  } as typeof DatabaseSync.prototype.prepare;
  StatementSync.prototype.all = function (this: InstanceType<typeof StatementSync>, ...args: unknown[]) {
    const sql = this.sourceSQL;
    const db = dbOf.get(this);
    if (sql.includes('ORDER BY json_extract') && db !== undefined) calls.push({ db, sql, args });
    return (originalAll as (...a: unknown[]) => unknown).apply(this, args);
  } as typeof StatementSync.prototype.all;
  try {
    const dir = await mkdtemp(join(tmpdir(), 'lead-plan-deep-'));
    dirs.push(dir);
    const backend = createSqliteRecordBackend({ path: join(dir, 'deep.sqlite'), schemas });
    const records = backend.records as unknown as RecordOrderedLookupPort;
    await records.lookupOrdered({ index: S2A_REQUEUE_BY_PROJECT_LOOKUP, values: ['P'], direction: 'asc', limit: 1 } as never);
    const db = calls.at(-1)?.db;
    if (db === undefined) throw new Error('no database handle');
    const N = 100_000;
    const pad = (n: number, w: number) => String(n).padStart(w, '0');
    const requeueRef = (i: number) => ({ aggregateType: 'TaskRequeue', projectId: 'P', goalId: 'G', taskId: 'T', requeueId: pad(i, 6) });
    const widgetRef = (i: number) => ({ aggregateType: 'Widget', projectId: 'P', widgetId: pad(i, 6) });
    const insert = originalPrepare.call(db, 'INSERT INTO snapshots (ref_key, snapshot_json) VALUES (?, ?)');
    db.exec('BEGIN');
    for (let i = 0; i < N; i += 1) {
      insert.run(canonicalJson(requeueRef(i)), JSON.stringify({ ref: requeueRef(i), since: `c${pad(i, 10)}` }));
      insert.run(canonicalJson(widgetRef(i)), JSON.stringify({ ref: widgetRef(i), group: 'g', rank: i }));
    }
    db.exec('COMMIT');
    const time = (sql: string, args: unknown[]) => {
      const statement = originalPrepare.call(db, sql);
      const t0 = process.hrtime.bigint();
      let rows = 0;
      for (let k = 0; k < 5; k += 1) rows = (originalAll as (...a: unknown[]) => unknown[]).apply(statement, args).length;
      return { ms: Number(process.hrtime.bigint() - t0) / 5e6, rows };
    };
    const lines: string[] = [];
    const cases = [
      { index: S2A_REQUEUE_BY_PROJECT_LOOKUP, values: ['P'], order: (i: number) => `c${pad(i, 10)}`, key: (i: number) => canonicalJson(requeueRef(i)) },
      { index: integerIndex.name, values: ['g'], order: (i: number) => i, key: (i: number) => canonicalJson(widgetRef(i)) },
    ];
    for (const c of cases) {
      for (const direction of ['asc', 'desc'] as const) {
        const deep = direction === 'asc' ? N - 300 : 300;
        const first = calls.length;
        await records.lookupOrdered({ index: c.index, values: c.values, direction, limit: 200 } as never);
        await records.lookupOrdered({ index: c.index, values: c.values, direction, limit: 200,
          after: { order: c.order(deep), refKey: c.key(deep) } } as never);
        await records.lookupOrdered({ index: c.index, values: c.values, direction, limit: 200,
          bound: { value: c.order(direction === 'asc' ? 0 : N - 1), inclusive: true },
          after: { order: c.order(deep), refKey: c.key(deep) } } as never);
        const [plain, after, both] = calls.slice(first, first + 3);
        if (plain === undefined || after === undefined || both === undefined) throw new Error('missing captured calls');
        const p = time(plain.sql, plain.args), a = time(after.sql, after.args), b = time(both.sql, both.args);
        lines.push(`${c.index} ${direction}: first ${p.ms.toFixed(2)} ms (${p.rows} rows), deep after ${a.ms.toFixed(2)} ms (${a.rows}), deep bound+after ${b.ms.toFixed(2)} ms (${b.rows})`);
        console.log(lines.at(-1));
        expect(a.rows).toBe(201);
        expect(b.rows).toBe(201);
        expect(a.ms).toBeLessThan(Math.max(1, p.ms * 5));
        expect(b.ms).toBeLessThan(Math.max(1, p.ms * 5));
      }
    }
    console.log(lines.join('\n'));
    await backend.close();
  } finally {
    StatementSync.prototype.all = originalAll;
  }
}, 600_000);
