/**
 * P1-02 Control entry: governance install (CompletionPolicy /
 * ArchitectureBaseline immutable revisions).
 *
 * ENTRY FILE (shared baseline — exported signature FROZEN). Lane A fills the
 * implementation; the semantics it must implement are recorded in
 * IMPLEMENTATION-HANDOFF.md "P1-02 契约与存储语义" and the ticket Acceptance:
 *   schema -> digest (JCS+SHA-256 over the whole fixture) -> fold the exact
 *   install commit (one event + one immutable revision snapshot, CAS at
 *   revision 0) -> map the ledger receipt (replay replayed=true, conflict ->
 *   revision_conflict, zero-write on every rejection).
 *   - missing/invalid fixture -> "invalid" (ZERO write); digest mismatch ->
 *     "digest_mismatch";
 *   - install NEVER auto-activates; NEVER creates an ArchitectureEvolutionPolicy.
 *
 * Guard style mirrors control-engine.ts: engine-level guards live only where a
 * distinct rejection code is required (validation -> invalid, digest ->
 * digest_mismatch). Immutability / idempotent replay / CAS are decided by
 * StateLedger.commit and mapped back here.
 *
 * Dependencies: only the frozen contracts (governance.js, ledger.js),
 * StateLedger.load/commit via ControlEngineDeps, the shared validation
 * (validation.js). The deterministic fold is built inline and verified by the
 * lane tests against the shared fixture builders (fold-equality).
 */
import type {
  ArchitectureBaselineInstalledEvent,
  ArchitectureBaselineRevisionRef,
  ArchitectureBaselineRevisionSnapshot,
  CompletionPolicyInstalledEvent,
  CompletionPolicyRevisionRef,
  CompletionPolicyRevisionSnapshot,
  GovernanceInstallCommand,
  GovernanceInstallReceipt,
  InstallArchitectureBaselineRevisionCommand,
  InstallCompletionPolicyRevisionCommand,
} from "../../contracts/governance.js";
import {
  governanceContentDigest,
  governanceInstallFingerprint,
} from "../../contracts/governance.js";
import type { GovernanceInstallLedgerCommitV1, LedgerCommitReceipt } from "../../contracts/ledger.js";
import {
  validateArchitectureBaselineFixture,
  validateCompletionPolicyFixture,
  validateInstallArchitectureBaselineRevisionCommand,
  validateInstallCompletionPolicyRevisionCommand,
} from "../../contracts/validation.js";
import type { ControlEngineDeps } from "./control-engine.js";

export function installGovernanceRevision(
  deps: ControlEngineDeps,
  command: GovernanceInstallCommand,
): Promise<GovernanceInstallReceipt> {
  if (command.commandType === "InstallCompletionPolicyRevision") {
    return installCompletionPolicyRevision(deps, command);
  }
  return installArchitectureBaselineRevision(deps, command);
}

// ------------------------------------------------------------------------ //
// CompletionPolicy revision                                                  //
// ------------------------------------------------------------------------ //

async function installCompletionPolicyRevision(
  deps: ControlEngineDeps,
  command: InstallCompletionPolicyRevisionCommand,
): Promise<GovernanceInstallReceipt> {
  // 1) schema: command-level validator DISCARDS fixture issues, so validate the
  //    fixture separately (a malformed fixture is an invalid command — zero-write).
  const issues = validateInstallCompletionPolicyRevisionCommand(command);
  const fixtureIssues = validateCompletionPolicyFixture(command.payload.fixture);
  const structural = issues.concat(fixtureIssues).filter((i) => i.code !== "digest_mismatch");
  // Semantic validity: a CompletionPolicy that recognizes NO requirement kind is
  // degenerate ("fixture 无效 -> invalid" per the frozen semantics).
  const emptyRequirementKinds = command.payload.fixture.content.requirementKinds.length === 0;
  if (structural.length > 0 || emptyRequirementKinds) {
    return { status: "rejected", commandId: command.commandId, code: "invalid" };
  }

  // 2) digest: pure content mismatch -> digest_mismatch (structural issues win).
  if (governanceContentDigest(command.payload.fixture) !== command.payload.contentDigest) {
    return { status: "rejected", commandId: command.commandId, code: "digest_mismatch" };
  }

  // 3) deterministic fold (exactly the shared fixture-fold builder).
  const batch = buildInstallCommit(command, deps);
  const revisionRef = batch.snapshots[0]!.ref as CompletionPolicyRevisionRef;

  // 4) atomic commit (idempotency / CAS by the ledger).
  const receipt = await deps.ledger.commit(batch);
  return mapInstallReceipt(receipt, command, revisionRef);
}

// ------------------------------------------------------------------------ //
// ArchitectureBaseline revision                                              //
// ------------------------------------------------------------------------ //

async function installArchitectureBaselineRevision(
  deps: ControlEngineDeps,
  command: InstallArchitectureBaselineRevisionCommand,
): Promise<GovernanceInstallReceipt> {
  const issues = validateInstallArchitectureBaselineRevisionCommand(command);
  const fixtureIssues = validateArchitectureBaselineFixture(command.payload.fixture);
  const structural = issues.concat(fixtureIssues).filter((i) => i.code !== "digest_mismatch");
  if (structural.length > 0) {
    return { status: "rejected", commandId: command.commandId, code: "invalid" };
  }

  if (governanceContentDigest(command.payload.fixture) !== command.payload.contentDigest) {
    return { status: "rejected", commandId: command.commandId, code: "digest_mismatch" };
  }

  const batch = buildInstallCommit(command, deps);
  const revisionRef = batch.snapshots[0]!.ref as ArchitectureBaselineRevisionRef;

  const receipt = await deps.ledger.commit(batch);
  return mapInstallReceipt(receipt, command, revisionRef);
}

// ------------------------------------------------------------------------ //
// Mapping: ledger receipt -> governance install receipt                       //
// ------------------------------------------------------------------------ //

function mapInstallReceipt(
  receipt: LedgerCommitReceipt,
  command: GovernanceInstallCommand,
  revisionRef: CompletionPolicyRevisionRef | ArchitectureBaselineRevisionRef,
): GovernanceInstallReceipt {
  if (receipt.status === "committed") {
    return {
      status: "committed",
      commandId: command.commandId,
      replayed: receipt.replayed,
      revisionRef,
      contentDigest: command.payload.contentDigest,
      eventIds: receipt.eventIds,
      commitCursor: receipt.commitCursor,
    };
  }
  switch (receipt.code) {
    case "invalid_commit":
      return { status: "rejected", commandId: command.commandId, code: "invalid" };
    case "revision_conflict":
      return { status: "rejected", commandId: command.commandId, code: "revision_conflict" };
    case "idempotency_conflict":
      return { status: "rejected", commandId: command.commandId, code: "idempotency_conflict" };
    case "unavailable":
      return { status: "rejected", commandId: command.commandId, code: "unavailable" };
    case "not_empty":
      // Not reachable for a governance-install commit (expectedVersions
      // non-empty); treat as a malformed commit rather than inventing a code.
      return { status: "rejected", commandId: command.commandId, code: "invalid" };
  }
}

// ------------------------------------------------------------------------ //
// Deterministic fold (replicates the shared fixture-fold builders exactly)    //
// ------------------------------------------------------------------------ //

function buildInstallCommit(
  command: GovernanceInstallCommand,
  deps: ControlEngineDeps,
): GovernanceInstallLedgerCommitV1 {
  const eventId = deps.eventId();
  const occurredAt = deps.now();

  if (command.commandType === "InstallCompletionPolicyRevision") {
    const fixture = command.payload.fixture;
    const ref: CompletionPolicyRevisionRef = {
      aggregateType: "CompletionPolicyRevision",
      projectId: command.identity.projectId,
      policyId: fixture.identity.policyId,
      revision: fixture.revision,
    };
    const event: CompletionPolicyInstalledEvent = {
      eventId,
      eventType: "CompletionPolicyInstalled",
      schemaVersion: 1,
      projectId: command.identity.projectId,
      aggregateType: "CompletionPolicyRevision",
      aggregateId: fixture.identity.policyId,
      aggregateRevision: 1,
      causationId: command.commandId,
      correlationId: command.correlationId,
      idempotencyKey: command.identity.idempotencyKey,
      actor: { ...command.identity.actor },
      occurredAt,
      payload: {
        policyId: fixture.identity.policyId,
        revision: fixture.revision,
        contentDigest: command.payload.contentDigest,
        content: fixture.content,
      },
    };
    const snapshot: CompletionPolicyRevisionSnapshot = {
      ref,
      revision: 1,
      schemaVersion: 1,
      policyId: fixture.identity.policyId,
      contentRevision: fixture.revision,
      contentDigest: command.payload.contentDigest,
      content: fixture.content,
    };
    return {
      commitKind: "governance-install",
      schemaVersion: 1,
      identity: { ...command.identity },
      fingerprint: governanceInstallFingerprint(command),
      expectedVersions: [{ ref, revision: 0 }],
      events: [event],
      snapshots: [snapshot],
      outboxIntents: [],
    };
  }

  const fixture = command.payload.fixture;
  const ref: ArchitectureBaselineRevisionRef = {
    aggregateType: "ArchitectureBaselineRevision",
    projectId: command.identity.projectId,
    baselineId: fixture.identity.baselineId,
    revision: fixture.revision,
  };
  const event: ArchitectureBaselineInstalledEvent = {
    eventId,
    eventType: "ArchitectureBaselineInstalled",
    schemaVersion: 1,
    projectId: command.identity.projectId,
    aggregateType: "ArchitectureBaselineRevision",
    aggregateId: fixture.identity.baselineId,
    aggregateRevision: 1,
    causationId: command.commandId,
    correlationId: command.correlationId,
    idempotencyKey: command.identity.idempotencyKey,
    actor: { ...command.identity.actor },
    occurredAt,
    payload: {
      baselineId: fixture.identity.baselineId,
      revision: fixture.revision,
      contentDigest: command.payload.contentDigest,
      content: fixture.content,
    },
  };
  const snapshot: ArchitectureBaselineRevisionSnapshot = {
    ref,
    revision: 1,
    schemaVersion: 1,
    baselineId: fixture.identity.baselineId,
    contentRevision: fixture.revision,
    contentDigest: command.payload.contentDigest,
    content: fixture.content,
  };
  return {
    commitKind: "governance-install",
    schemaVersion: 1,
    identity: { ...command.identity },
    fingerprint: governanceInstallFingerprint(command),
    expectedVersions: [{ ref, revision: 0 }],
    events: [event],
    snapshots: [snapshot],
    outboxIntents: [],
  };
}
