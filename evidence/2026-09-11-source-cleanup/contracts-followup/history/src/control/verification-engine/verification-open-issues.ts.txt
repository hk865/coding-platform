/**
 * VerificationEngine 的「未处置问题」出口。
 *
 * 为什么由 VerificationEngine 提供：验证结论（工具轮次与独立审阅）是本模块的持久
 * 事实；返工提案只能消费这些结论，不能自己重新判断验收是否通过。因此这里只做
 * 「读已提交事实 + 归一成问题」，不做判定、不写状态、不启动 Agent。
 *
 * 两类来源都不改写：
 *   1. 工具轮次的 coverage 里 result 非 PASS 的要求，来自本模块持久 journal；
 *   2. 独立审阅逐条结论里 outcome 非 PASS 的要求，来自本模块已经正式登记结果的
 *      审阅记录（resultReceipt 已受理，重开后由持久记录重建）。
 *
 * 每条失败要求都带结构化失败事实（ReworkFailureFactV1）：分类、命令、退出码、是否超时、
 * 耗时、有界 stderr 摘要、来源摘要与原始报告引用。为什么要有这些：只返回「FAIL」不能回答
 * 用户与返工角色最先要问的「为什么 fail」。分类一律复用 CommandCheckProvider 已经持久化的
 * category，本文件不产生第二套分类，也不从失败文本反推原因。
 *
 * currentness 只表达「这条问题是否还属于当前 active plan revision」：
 *   - open：问题来自当前 active revision，尚未被新 revision 取代；
 *   - superseded：Goal 已经切到别的 revision（例如已受理的返工计划）；
 *   - unknown：无法解析 Goal 或 active revision 时不猜测，明确标注。
 *
 * RC-01 起，每条问题另外带一份**处置事实**（disposition），与 currentness 正交：
 * currentness 判的是「证据 anchor 是否仍生效」，disposition 判的是「这些失败义务有没有被处置」。
 * 为什么必须分开：一次触发受理分组 A 之后，分组 B 的问题 anchor 也会失效，但它既没有被任何
 * 返工任务接手、也没有在当前 revision 上重验通过——那是**未处置**，不是已被处置。旧实现只有
 * currentness 一条轴，于是这种情况只能被标成 superseded，失败义务随之凭空消失（RC-01）。
 * disposition 的判据全部来自 canonical 事实（当前 revision 的任务 disposition 与义务承担者、
 * 本模块 journal 里**锚定当前 revision** 的 PASS 结论），不猜、不从文本反推：
 *   - unaddressed：anchor 就是当前 revision，尚未处置；
 *   - carried_by_task：anchor 已失效，但当前 revision 里这些义务仍由原任务承担（它仍 active）
 *     且没有在当前 revision 上重验通过 → 仍未处置，落点明确；
 *   - disposed_by_rework：承担者已经换人（原任务不再承担这些义务，改由别的 active 任务承担）；
 *   - disposed_by_reverification：这些要求已在当前 revision 上重新验证通过；
 *   - unknown：读不到当前 revision 的计划（或问题声明的义务不在其中）→ 无法判断，不当作已处置。
 */

/**
 * 原始报告只读端口。与 VerificationRounds／CommandCheckLifecycle 使用同一个报告读取入口
 * （Context.openReport → ArtifactVault.open），因此权限、owner 与摘要校验仍是同一套。
 */
export type OpenIssuesReportReader = (ref: ArtifactRef, owner: RunRef) => Promise<ArtifactOpenResult>;

/** 只读投影：把两类已提交事实归一为问题，并标注它们相对当前 revision 的时效。 */
export class VerificationOpenIssues {
  constructor(private readonly deps: {
    journal: VerificationJournal;
    disposition: ReworkDispositionPort;
    /**
     * 可选：原始报告读取端口由模块入口注入。未注入时投影不会假装字段存在，
     * 而是在该条事实的 gaps 中写明「执行细节需要按 reportRef 读取」。
     */
    reports?: OpenIssuesReportReader;
  }) {}

  async openIssues(request: OpenIssuesRequestV1): Promise<OpenIssuesViewV1> {
    const gaps: string[] = [];
    const raw = [...await this.roundIssues(request, gaps), ...this.reviewIssues(request, gaps)];
    const issues = await this.deps.disposition.projectIssues(request, raw);

    // 同一任务同一问题的确定性排序：先按任务，再按问题身份，保证视图可重建。
    issues.sort((a, b) => (a.taskId === b.taskId ? a.issueId.localeCompare(b.issueId) : a.taskId.localeCompare(b.taskId)));
    if (issues.length > REWORK_MAX_ISSUES)
      return { status: 'unavailable', code: 'unavailable', message: '未处置问题超过 ' + REWORK_MAX_ISSUES + ' 条；请先收敛范围或分批返工' };
    if (!issues.length) return { status: 'none', gaps };
    return { status: 'ready', issues, gaps };
  }

  private coveredTasks(request: OpenIssuesRequestV1, taskId: string): boolean {
    return request.taskIds.length === 0 || request.taskIds.includes(taskId);
  }

  /** 工具轮次：只取本 Goal 范围内、结论非 PASS 的轮次里的失败覆盖项。 */
  private async roundIssues(request: OpenIssuesRequestV1, gaps: string[]): Promise<ReworkIssueViewV1[]> {
    const issues: ReworkIssueViewV1[] = [];
    for (const round of this.deps.journal.rounds) {
      const scope = round.scope;
      if (scope.projectId !== request.projectId || scope.workspaceId !== request.workspaceId || scope.goalId !== request.goalId) continue;
      if (!this.coveredTasks(request, scope.taskId)) continue;
      if (round.outcome === 'PASS' || round.outcome === null) continue;
      // 只有「检查给出了否定结论」才是返工能解决的问题。result=null 表示本次轮次根本没有为该要求
      // 产出结论（例如 reviewer 类要求、或注册配置没有覆盖它），那是材料/覆盖缺口，不是实现缺陷；
      // 把它算成失败会让整批问题带上一条无法返工的要求，从而正确地被自动受理边界拒绝。
      const failed = round.coverage.filter((entry) => entry.result === 'FAIL' || entry.result === 'INCONCLUSIVE');
      const uncovered = round.coverage.filter((entry) => entry.result === null);
      if (!failed.length) {
        gaps.push('轮次 ' + round.roundId + ' 结论为 ' + String(round.outcome) + '，但没有给出否定结论的覆盖项（' +
          (uncovered.length ? uncovered.length + ' 项没有结论：' + uncovered.map(entry => entry.obligationId + '/' + entry.requirementId).join('、') : '没有覆盖项') +
          '）；该轮次不产生返工问题，缺项应由覆盖/材料路径处理。');
        continue;
      }
      if (uncovered.length)
        gaps.push('轮次 ' + round.roundId + ' 另有 ' + uncovered.length + ' 项要求没有结论（' +
          uncovered.map(entry => entry.obligationId + '/' + entry.requirementId).join('、') + '）；它们不属于本次返工范围。');
      // 只为会被返回的条目读取原始报告：确认事实需要有界 I/O，截断的部分不生成半份事实。
      const covered = failed.slice(0, REWORK_MAX_FAILED_REQUIREMENTS);
      if (covered.length < failed.length)
        gaps.push('问题 ' + scope.taskId + ' 的失败要求超过 ' + REWORK_MAX_FAILED_REQUIREMENTS + ' 条；本次只为前 ' + REWORK_MAX_FAILED_REQUIREMENTS + ' 条读取原始报告，其余条目请分批返工。');
      const failedRequirements: ReworkFailedRequirementV1[] = [];
      for (const entry of covered) failedRequirements.push(await this.roundRequirement(round, entry, round.outcome));
      issues.push(this.buildIssue({
        request,
        taskId: scope.taskId,
        runRef: { aggregateType: 'Run', projectId: request.projectId, goalId: request.goalId, runId: scope.runId },
        planRef: round.materialIdentity?.planRef ?? null,
        planRevision: round.materialIdentity?.planRevision ?? 0,
        failedRequirements,
        evidenceRefs: this.roundEvidenceRefs(round.aggregate?.admissions ?? [], request.projectId),
        source: {
          kind: 'verification_round',
          roundId: round.roundId,
          requestId: round.requestId,
          outcome: round.outcome === 'INCONCLUSIVE' ? 'INCONCLUSIVE' : 'FAIL',
          status: round.status === 'interrupted' ? 'interrupted' : 'completed',
          sourceDigest: round.materialIdentity?.sourceDigest ?? null,
        },
        summary: '工具验证轮次 ' + round.roundId + ' 在任务 ' + scope.taskId + ' 上有 ' + failed.length + ' 项验收要求未通过（结论 ' + round.outcome + '）。',
        detectedAt: round.finishedAt ?? round.createdAt,
      }, gaps));
    }
    return issues;
  }

  private async roundRequirement(round: VerificationRoundRecord, entry: VerificationRoundCoverage, outcome: string): Promise<ReworkFailedRequirementV1> {
    const failure = await this.roundFailure(round, entry);
    return {
      obligationId: entry.obligationId,
      requirementId: entry.requirementId,
      kind: entry.kind,
      reason: this.failureReason(entry, outcome, failure),
      failure,
    };
  }

  /**
   * 工具轮次里一条失败要求的结构化事实。
   *
   * 事实分两段取，因为它们的持久位置不同：
   *   - 分类与结论在检查记录里（progress 的 report_stored 阶段由 CommandCheckProvider 写入）；
   *   - 退出码、是否超时、stderr、精确耗时只在原始报告正文里，需要报告读取端口。
   * 任何一段取不到都记入 gaps，不做推断，也不影响该条要求仍是「未通过」。
   */
  private async roundFailure(round: VerificationRoundRecord, entry: VerificationRoundCoverage): Promise<ReworkFailureFactV1> {
    if (entry.kind === 'reviewer')
      return this.untooledFailure('该要求是 reviewer 类，由独立审阅的逐条结论承担；工具轮次不产生命令检查分类', 'reviewer', null);
    // 只按轮次已经登记的检查身份查记录；查不到就是没有留下记录，不用别的轮次或别的检查顶替。
    const items = entry.checkIds.map((checkId) => ({ checkId, check: this.child(round, checkId) }));
    if (!items.length)
      return this.untooledFailure('轮次没有为该要求登记工具检查（覆盖缺失，不是某条命令的结论）', entry.kind, round.materialIdentity?.sourceDigest ?? null);
    const failedCheckIds = items.filter((candidate) => this.checkResult(candidate.check) !== 'PASS').map((candidate) => candidate.checkId).sort();
    // 一条要求可能由多条检查共同覆盖：优先用结论为 FAIL 的一条作主事实，其次用没有结论的一条。
    // 其余未通过的检查身份全部列在 failedCheckIds 里，避免只报一条造成误解。
    const decisive = items.find((candidate) => this.checkResult(candidate.check) === 'FAIL')
      ?? items.find((candidate) => this.checkResult(candidate.check) !== 'PASS')
      ?? items[0]!;
    return this.toolFailure(round, decisive.checkId, decisive.check, failedCheckIds, entry.kind);
  }

  private async toolFailure(round: VerificationRoundRecord, checkId: string, check: CommandCheckRecord | null, failedCheckIds: string[], coverageKind: 'static' | 'dynamic'): Promise<ReworkFailureFactV1> {
    const gaps: string[] = [];
    const progress = check?.progress ?? null;
    const stored = progress?.phase === 'report_stored' ? progress : null;
    const observation = check?.result?.status === 'ready' ? check.result.observations[0] ?? null : null;
    // 种类优先取检查自身的注册定义；取不到时回退到覆盖项声明的种类，而不是默认值。
    const kind = check?.roundBinding?.definition.kind ?? round.checks.find((item) => item.definition.checkId === checkId)?.definition.kind ?? coverageKind;
    // 已持久化的检查结论：报告阶段写入的值优先，其次是从结果恢复的观察。
    const recorded = stored?.result ?? observation?.result ?? null;
    const fact: ReworkFailureFactV1 = {
      checkId,
      failedCheckIds,
      kind,
      category: stored?.category ?? null,
      result: recorded === 'PASS' ? null : recorded,
      command: check?.command ?? null,
      timeoutMs: check?.timeoutMs ?? null,
      exitCode: null,
      timedOut: null,
      durationMs: elapsedMs(check),
      stderrExcerpt: null,
      sourceDigest: progress?.sourceDigest ?? round.materialIdentity?.sourceDigest ?? null,
      reportRef: stored?.artifactRef ?? observation?.artifactRef ?? null,
      gaps,
    };
    if (!check) {
      gaps.push('检查 ' + checkId + ' 没有持久记录（身份 ' + checkId + '）；分类与执行结果都无法给出');
      return fact;
    }
    if (!stored)
      gaps.push('检查 ' + checkId + ' 没有已落盘的报告阶段记录（状态 ' + check.status + '，生命周期 ' + (check.lifecycle ?? '未记录') + '）；分类与执行结果不可用');
    if (fact.category === null) gaps.push('检查 ' + checkId + ' 没有持久化的失败分类；不推测分类');
    else if (!(COMMAND_CHECK_CATEGORIES as readonly string[]).includes(fact.category))
      gaps.push('检查 ' + checkId + ' 的分类 ' + fact.category + ' 不在本契约收录的提供者取值内；按原值保留，不映射成别的分类');
    if (recorded === 'PASS')
      gaps.push('检查 ' + checkId + ' 的记录结论是 PASS，与该要求未通过的覆盖结论不一致；本事实不采用该结论，请按原始报告核对');
    else if (fact.result === null) gaps.push('检查 ' + checkId + ' 没有产生结论；该要求仍未通过，不是通过');
    if (fact.command === null) gaps.push('检查 ' + checkId + ' 没有登记命令原文');
    if (fact.reportRef === null) gaps.push('检查 ' + checkId + ' 没有原始报告引用；执行细节（exitCode／timedOut／stderrExcerpt）没有来源');
    else await this.readExecution(fact, round.materialIdentity?.runRef ?? null, {
      observationId: progress?.observationId ?? null, result: fact.result, category: fact.category,
    });
    return fact;
  }

  /**
   * 从原始报告补齐执行细节。只有报告身份与该检查记录逐项一致时才采用其中的值：
   * 把别的执行的退出码挂到本问题上，比不报退出码更糟。
   */
  private async readExecution(fact: ReworkFailureFactV1, owner: RunRef | null, expected: { observationId: string | null; result: ReworkFailureFactV1['result']; category: string | null }): Promise<void> {
    const ref = fact.reportRef!;
    if (owner === null) {
      fact.gaps.push('缺少该检查所属 Run 的身份；不能按 owner 授权读取原始报告 ' + shortRef(ref));
      return;
    }
    if (!this.deps.reports) {
      fact.gaps.push('本投影没有注入原始报告读取端口；exitCode／timedOut／stderrExcerpt 只能按 reportRef ' + shortRef(ref) + ' 从 ArtifactVault 读取');
      return;
    }
    const opened = await this.deps.reports(ref, owner);
    if (opened.status !== 'ready') {
      fact.gaps.push('原始报告 ' + shortRef(ref) + ' 当前不可读取（' + opened.status + '，可能已被清理或权限不符）；执行细节无事实可给，该要求仍判为未通过');
      return;
    }
    const body = opened.record.body;
    if (artifactBodyDigest(body) !== ref.digest || artifactBodySize(body) !== ref.sizeBytes) {
      fact.gaps.push('原始报告正文与引用摘要不一致；拒绝使用该正文，执行细节按不可用处理');
      return;
    }
    let report: Record<string, unknown>;
    try { report = JSON.parse(body) as Record<string, unknown>; }
    catch { fact.gaps.push('原始报告正文不是可解析的 JSON；执行细节按不可用处理'); return; }
    if (report['schemaVersion'] !== 1) { fact.gaps.push('原始报告版本 ' + String(report['schemaVersion']) + ' 不受支持；执行细节按不可用处理'); return; }
    // 同一性核对：报告必须就是这条检查这一次执行留下的正文，身份对不上就整份不用。
    if (expected.observationId !== null && report['observationId'] !== expected.observationId) {
      fact.gaps.push('原始报告的观察身份与检查记录不一致；执行细节按不可用处理');
      return;
    }
    if (report['result'] !== expected.result || report['category'] !== expected.category) {
      fact.gaps.push('原始报告的结果或分类与检查记录不一致；执行细节按不可用处理');
      return;
    }
    const execution = report['execution'];
    if (!execution || typeof execution !== 'object' || Array.isArray(execution)) {
      fact.gaps.push('原始报告没有命令执行记录（该分类下命令没有执行）；exitCode／timedOut／stderrExcerpt 无事实可给');
      return;
    }
    const exec = execution as Record<string, unknown>;
    // 每个字段只在报告确实记了它的时候才写值：null 是「报告说没有退出码」（例如被信号终止），
    // 与「报告压根没记」是两件事，后者必须留下缺口。
    if (typeof exec['exitCode'] === 'number' || exec['exitCode'] === null) fact.exitCode = exec['exitCode'] as number | null;
    else fact.gaps.push('原始报告没有记录退出码');
    if (typeof exec['timedOut'] === 'boolean') fact.timedOut = exec['timedOut'];
    else fact.gaps.push('原始报告没有记录是否超时');
    const stderr = exec['stderr'];
    if (stderr && typeof stderr === 'object' && !Array.isArray(stderr)) {
      const text = (stderr as Record<string, unknown>)['text'];
      if (typeof text === 'string') {
        const bounded = boundedText(text, REWORK_FAILURE_STDERR_MAX_BYTES);
        fact.stderrExcerpt = bounded.text;
        if (bounded.truncated)
          fact.gaps.push('stderr 摘要按 ' + REWORK_FAILURE_STDERR_MAX_BYTES + ' 字节截断；完整正文见 reportRef ' + shortRef(ref));
      } else fact.gaps.push('原始报告没有记录 stderr 文本');
    } else fact.gaps.push('原始报告没有记录 stderr');
    const timings = exec['timings'];
    if (timings && typeof timings === 'object' && !Array.isArray(timings)) {
      const executionMs = (timings as Record<string, unknown>)['executionMs'];
      // 报告的执行计时比记录的起止时间精确，读到就覆盖。
      if (typeof executionMs === 'number' && Number.isFinite(executionMs)) fact.durationMs = executionMs;
    }
  }

  /** 没有工具检查可归因的要求（reviewer 类、覆盖缺失）：不给分类，只说明它为什么没有分类。 */
  private untooledFailure(reason: string, kind: ReworkFailureFactV1['kind'], sourceDigest: string | null): ReworkFailureFactV1 {
    return {
      checkId: null,
      failedCheckIds: [],
      kind,
      category: null,
      result: null,
      command: null,
      timeoutMs: null,
      exitCode: null,
      timedOut: null,
      durationMs: null,
      stderrExcerpt: null,
      sourceDigest,
      reportRef: null,
      gaps: [reason],
    };
  }

  /** 人可读的失败原因：全部由结构化事实拼出，不额外判断分类，也不复制报告正文。 */
  private failureReason(entry: VerificationRoundCoverage, outcome: string, failure: ReworkFailureFactV1): string {
    if (failure.checkId === null)
      return failure.kind === 'reviewer'
        ? '该要求由独立审阅承担，工具轮次没有结论（轮次结论 ' + outcome + '）；逐条结论与结果身份见本问题的来源与审阅结果。'
        : '没有注册检查覆盖该要求（轮次结论 ' + outcome + '）；失败来自覆盖缺失，不是某条命令的结论。';
    const parts = ['命令检查 ' + failure.checkId + '：分类 ' + (failure.category ?? '未记录') + '，结论 ' + (failure.result ?? '未产生')];
    if (failure.command) parts.push('命令「' + failure.command + '」');
    if (failure.exitCode !== null) parts.push('退出码 ' + failure.exitCode);
    else if (failure.timedOut === true) parts.push('被超时终止，没有退出码');
    if (failure.timedOut === true) parts.push('超时预算 ' + (failure.timeoutMs ?? '未记录') + 'ms ' + (failure.durationMs !== null ? '（实际耗时 ' + failure.durationMs + 'ms）' : ''));
    else if (failure.durationMs !== null) parts.push('耗时 ' + failure.durationMs + 'ms');
    if (failure.stderrExcerpt) parts.push('stderr：' + firstLine(failure.stderrExcerpt, 120));
    const others = failure.failedCheckIds.filter((checkId) => checkId !== failure.checkId);
    if (others.length) parts.push('同一要求下未通过的检查还有 ' + others.join('、'));
    const reason = parts.join('；').replace(/；\s*$/, '');
    return reason + '。' + (failure.gaps.length ? '（事实缺口 ' + failure.gaps.length + ' 项，见 failure.gaps；缺口不代表通过）' : '');
  }

  private child(round: VerificationRoundRecord, checkId: string): CommandCheckRecord | null {
    const item = round.checks.find((candidate) => candidate.definition.checkId === checkId);
    if (!item) return null;
    // 与 VerificationRounds.child 同一身份规则：同为该轮次请求身份、同作用域的检查记录。
    return this.deps.journal.checks.find((check) => check.requestId === item.requestId && canonicalJson(vScope(check)) === canonicalJson(vScope(round.scope))) ?? null;
  }

  /** 检查结论只读已持久化的值：报告阶段写入的分类结论优先，其次是从结果恢复的观察。 */
  private checkResult(check: CommandCheckRecord | null): 'PASS' | 'FAIL' | 'INCONCLUSIVE' | null {
    if (!check) return null;
    if (check.progress?.phase === 'report_stored') return check.progress.result;
    return check.result?.status === 'ready' ? check.result.observations[0]?.result ?? null : null;
  }

  private roundEvidenceRefs(admissions: Array<{ evidenceId: string; status: string }>, projectId: string): EvidenceRef[] {
    return admissions.filter((entry) => entry.status === 'admitted').map((entry) => ({ aggregateType: 'Evidence' as const, projectId, evidenceId: entry.evidenceId }));
  }

  /**
   * 独立审阅：只取「已经正式登记结果」的审阅（resultReceipt 为 accepted/replayed），
   * 且结论为 accepted。报告资格被拒或结果未被正式登记的审阅不构成返工问题——
   * 那说明需要重做的是审阅材料，而不是被审工作。
   *
   * 这里不复制审阅的 Evidence 索引：正式证据归属 Control 的账本，问题只需指出
   * 「哪次审阅的哪些要求没通过」；提案阶段再由 Control 解析对应证据。
   */
  private reviewIssues(request: OpenIssuesRequestV1, gaps: string[]): ReworkIssueViewV1[] {
    const issues: ReworkIssueViewV1[] = [];
    for (const record of this.deps.journal.reviews) {
      const scope = record.scope;
      if (scope.projectId !== request.projectId || scope.workspaceId !== request.workspaceId || scope.goalId !== request.goalId) continue;
      if (!this.coveredTasks(request, scope.taskId)) continue;
      const receipt = record.resultReceipt;
      if (receipt === null || receipt.status === 'rejected') {
        gaps.push('审阅 ' + record.reviewId + ' 的结果尚未被正式登记，不构成返工问题。');
        continue;
      }
      const command = record.resultCommand;
      if (command === null || command.decision.status !== 'accepted') {
        gaps.push('审阅 ' + record.reviewId + ' 的结论未被接受，不构成返工问题。');
        continue;
      }
      const failed = command.decision.requirements.filter((entry) => entry.outcome !== 'PASS');
      if (!failed.length) continue;
      issues.push(this.buildIssue({
        request,
        taskId: scope.taskId,
        runRef: command.output.runRef,
        planRef: record.planRef,
        planRevision: 0,
        failedRequirements: failed.map((entry) => this.reviewRequirement(record, entry)),
        evidenceRefs: [],
        source: { kind: 'review_verdict', reviewId: scope.taskId + ':' + record.requestId, requestId: record.requestId, resultRef: receipt.resultRef?.reviewId ?? record.reviewId },
        summary: '独立审阅 ' + record.requestId + ' 在任务 ' + scope.taskId + ' 上有 ' + failed.length + ' 项验收要求未通过。',
        detectedAt: record.createdAt,
      }, gaps));
    }
    return issues;
  }

  /**
   * 独立审阅的一条失败要求：结论与结果身份沿用既有字段（outcome／summary／resultRef），
   * 不把它塞进命令检查分类，也不为它发明新分类。
   */
  private reviewRequirement(record: ReviewJournalRecord, entry: ReviewerRequirementResult): ReworkFailedRequirementV1 {
    const gaps: string[] = ['独立审阅要求没有命令检查分类：结论来自审阅的逐条判定，exitCode／timedOut／stderr 对它不适用'];
    if (record.descriptor === null) gaps.push('审阅记录没有留下材料描述；被审版本摘要不可用');
    if (record.resultCommand?.output.reportRef === undefined) gaps.push('审阅结果没有留下原始报告引用；逐条结论文本见本问题的 reason 与审阅结果身份');
    return {
      obligationId: entry.obligationId,
      requirementId: entry.requirementId,
      kind: 'reviewer',
      reason: '独立审阅结论 ' + entry.outcome + '：' + entry.summary,
      failure: {
        checkId: null,
        failedCheckIds: [],
        kind: 'reviewer',
        category: null,
        result: entry.outcome === 'PASS' ? null : entry.outcome,
        command: null,
        timeoutMs: null,
        exitCode: null,
        timedOut: null,
        durationMs: null,
        stderrExcerpt: null,
        sourceDigest: record.descriptor?.materialIdentity.sourceDigest ?? null,
        reportRef: record.resultCommand?.output.reportRef ?? null,
        gaps,
      },
    };
  }

  private buildIssue(input: {
    request: OpenIssuesRequestV1;
    taskId: string;
    runRef: RunRef;
    planRef: PlanRevisionRef | null;
    planRevision: number;
    failedRequirements: ReworkFailedRequirementV1[];
    evidenceRefs: EvidenceRef[];
    source: ReworkIssueSourceV1;
    summary: string;
    detectedAt: string;
  }, gaps: string[]): ReworkIssueViewV1 {
    const source = input.source;
    const failedRequirements = input.failedRequirements.slice(0, REWORK_MAX_FAILED_REQUIREMENTS);
    if (failedRequirements.length < input.failedRequirements.length)
      gaps.push('问题 ' + input.taskId + ' 的失败要求超过 ' + REWORK_MAX_FAILED_REQUIREMENTS + ' 条，已按上限截断；请分批返工。');
    const issueId = reworkIssueIdFor({ taskId: input.taskId, source, failedRequirements });
    return {
      schemaVersion: 1,
      issueId,
      projectId: input.request.projectId,
      workspaceId: input.request.workspaceId,
      goalId: input.request.goalId,
      taskId: input.taskId,
      runRef: input.runRef,
      planRef: input.planRef ?? { aggregateType: 'PlanRevision', projectId: input.request.projectId, planId: 'unresolved' },
      planRevision: input.planRevision,
      source,
      failedRequirements,
      evidenceRefs: input.evidenceRefs,
      summary: input.summary,
      detectedAt: input.detectedAt,
      currentness: { status: 'unknown', issues: [] },
      reworkTaskId: reworkTaskIdFor(issueId),
    };
  }

}

/** 检查记录里已经有的耗时近似；报告的执行计时更精确，读到时会覆盖它。 */
function elapsedMs(check: CommandCheckRecord | null): number | null {
  const start = check?.startedAt ? Date.parse(check.startedAt) : Number.NaN;
  const end = check?.finishedAt ? Date.parse(check.finishedAt) : Number.NaN;
  return Number.isFinite(start) && Number.isFinite(end) && end >= start ? end - start : null;
}

/** 摘要按字节截断（不是按字符），避免多字节文本被截成半个字符或超过上限。 */
function boundedText(value: string, maxBytes: number): { text: string; truncated: boolean } {
  if (Buffer.byteLength(value, 'utf8') <= maxBytes) return { text: value, truncated: false };
  let end = value.length;
  while (end > 0 && Buffer.byteLength(value.slice(0, end), 'utf8') > maxBytes) end--;
  return { text: value.slice(0, end), truncated: true };
}

function firstLine(value: string, cap: number): string {
  const line = value.split('\n').find((entry) => entry.trim().length > 0) ?? value;
  return line.length > cap ? line.slice(0, cap) + '…' : line;
}

function shortRef(ref: ArtifactRef): string {
  return ref.digest.slice(0, 12) + '（' + ref.sizeBytes + ' 字节）';
}
