import { afterEach, expect, it } from 'vitest';
import { mkdtemp, mkdir, writeFile, rm } from 'node:fs/promises';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { createPersistentSqliteHarness } from '../../src/harness/persistent-harness.js';
import { prepareDispatchScenario, buildPreparedClaim, buildPreparedEnvelope } from '../contract-suite/p1-03-harness.js';
import { buildInstallCommand, buildActivateCommand } from "../../src/contracts/fixtures/governance-fixtures.js";
import { architectureBaselinePinFor } from "../../src/contracts/governance.js";
import { DISPATCH_PLAN_REVISION_FIXTURE_V1, buildDispatchStartCommand, buildManifestFixture } from '../../src/contracts/fixtures/dispatch-fixtures.js';
import { buildApplyPlanCommand } from '../../src/contracts/fixtures/plan-fixtures.js';
import { workspaceProjectIndex, ProjectArchitectureSourceReader } from '../../src/data/workspace-reader/source-workspace-reader.js';
import { SourceGraphContextCompiler } from '../../src/data/context-compiler/source-graph-context.js';
import { captureArchitectureSource } from '../../src/data/workspace-reader/architecture-source.js';
import { ArchitectureReconcilerImpl } from '../../src/control/architecture-reconciler/architecture-reconciler.js';
import { ArchitectureContextCompiler } from '../../src/data/context-compiler/architecture-context-compiler.js';
import type { ArchitectureInspectionIntentV1, ArchitectureInspectionSnapshot } from '../../src/contracts/architecture-inspection.js';
import type { ArchitectureSourceMapping } from '../../src/contracts/architecture-source.js';
import { governanceContentDigest } from '../../src/contracts/governance.js';
import type { PlanRevisionSnapshot } from '../../src/contracts/plan.js';

const cleanup:Array<()=>Promise<void>>=[];
afterEach(async()=>{for(const work of cleanup.splice(0).reverse())await work();});
const at='2026-09-09T00:00:00.000Z';
const scope={projectId:'proj-alpha',workspaceId:'ws-shared'};

it('binds real TS sources to an immutable baseline, records distinct deltas, reopens bodies, rejects stale and mismatched inputs',async()=>{
 const directory=await mkdtemp(join(tmpdir(),'source-architecture-'));cleanup.push(()=>rm(directory,{recursive:true,force:true}));
 const root=join(directory,'workspace');await mkdir(join(root,'src'),{recursive:true});
 await writeFile(join(root,'tsconfig.json'),' {"compilerOptions":{"module":"esnext","moduleResolution":"bundler"},"include":["src/*.ts"]}');
 await writeFile(join(root,'src/data.ts'),'export const value = 1;');
 await writeFile(join(root,'src/query.ts'),'export const query = 1;');
 const mappings:ArchitectureSourceMapping[]=[{id:'Data',kind:'module',paths:['src/data.ts']},{id:'Query',kind:'module',paths:['src/query.ts']},{id:'DataAPI',kind:'interface',paths:['src/data.ts']}];
 const index=await workspaceProjectIndex(root);cleanup.push(async()=>index.dispose());
 const source=await captureArchitectureSource(index,{...scope,workspaceRevision:1},mappings);
 await mkdir(join(directory,'state'));
 let h=await createPersistentSqliteHarness({dir:join(directory,'state')});cleanup.push(()=>h.close());
 // The source baseline is explicit initial governance, installed before the plan.
 // Do not bypass the formal plan-revision protocol by replacing an already active plan.
 await prepareDispatchScenario({...h,submit:command=>h.control.submit(command),install:command=>{
   if(command.commandType==='InstallArchitectureBaselineRevision' && command.identity.projectId===scope.projectId){
     command.payload.fixture={...command.payload.fixture,content:{...command.payload.fixture.content,sourceBinding:source,dependencyRules:[{ruleId:'no-query-data',kind:'forbid_dependency',fromModule:'Query',toModule:'Data'}]}};
     command.payload.contentDigest=governanceContentDigest(command.payload.fixture);
   }
   return h.install(command);
 }});
 const planLoad=await h.ledger.load({aggregateType:'PlanRevision',projectId:scope.projectId,planId:DISPATCH_PLAN_REVISION_FIXTURE_V1.planId});
 if(planLoad.status!=='found')throw Error('plan missing');
 const plan=planLoad.snapshot as PlanRevisionSnapshot,applied={planRef:plan.ref},pin=plan.effectiveArchitectureBaseline;
 const claim=await h.claimTask(buildPreparedClaim({commandId:'source-reader',runId:'source-reader',attemptId:'source-reader',idempotencyKey:'source-reader'}));
 expect(claim).toMatchObject({status:'committed'});if(claim.status!=='committed')throw Error(JSON.stringify(claim));
 const put=await h.vault.put({body:'Source inspection input',contentType:'text/plain',ownerRef:claim.runRef,sourceRefs:[{kind:'plan-revision',refId:applied.planRef.planId,revision:'2'}],requestedAt:at});
 if(put.status!=='stored')throw Error('bundle missing');
 const envelope={...buildPreparedEnvelope({runId:'source-reader',attemptId:'source-reader',bundleRef:put.ref}),planRef:applied.planRef};
 expect(await h.startRun(buildDispatchStartCommand({projectId:scope.projectId,commandId:'start-source-reader',correlationId:'source',runId:'source-reader',submittedAt:at,envelope,manifest:buildManifestFixture({workspaceId:scope.workspaceId,workspaceRevision:1,planRef:applied.planRef})}))).toMatchObject({status:'committed'});
 const reader=new SourceGraphContextCompiler({ledger:()=>h.ledger,vault:()=>h.vault,source:new ProjectArchitectureSourceReader(()=>root),now:()=>at});
 const reconciler=()=>new ArchitectureReconcilerImpl({context:new ArchitectureContextCompiler({ledger:h.ledger,workspaceReader:reader}),vault:h.vault,control:h.control,now:()=>at});
 const intent:ArchitectureInspectionIntentV1={schemaVersion:1,inspectionId:'unchanged',...scope,workspaceRevision:1,planRef:applied.planRef,baselinePin:pin,source:'mechanic',requestedByRunRef:claim.runRef,reportInput:null,budget:{maxTokens:1,deadline:null}};
 expect(await reconciler().inspect(intent)).toMatchObject({status:'recorded',outcome:{findingCount:0}});
 await writeFile(join(root,'src/data.ts'),'export const value = 2;');
 const first={...intent,inspectionId:'data-edit'};
 const firstResult=await reconciler().inspect(first);expect(firstResult).toMatchObject({status:'recorded',outcome:{findingCount:2}});
 const load=async(id:string)=>{const row=await h.ledger.load({aggregateType:'ArchitectureInspection',...scope,inspectionId:id});if(row.status!=='found')throw Error('missing inspection');return row.snapshot as ArchitectureInspectionSnapshot;};
 const firstRow=await load(first.inspectionId);expect(firstRow.briefRef).not.toBeNull();expect(firstRow.proposalRef).toBeNull();
 const opened=await h.vault.open(firstRow.deltaRef!,{requesterRunRef:claim.runRef});expect(opened.status).toBe('ready');
 if(opened.status!=='ready')throw Error('missing delta');const firstDelta=JSON.parse(opened.record.body);expect(firstDelta.noVerdict).toBe(true);expect(firstDelta.changes.every((c:{structuralKey:string})=>c.structuralKey.includes('Data'))).toBe(true);
 await h.close();h=await h.reopen();
 expect(await h.vault.open(firstRow.deltaRef!,{requesterRunRef:claim.runRef})).toEqual(opened);
 expect(await reconciler().inspect(first)).toMatchObject({status:'recorded',outcome:{findingCount:2}});
 await writeFile(join(root,'src/data.ts'),'export const value = 1;');
 await writeFile(join(root,'src/query.ts'),'import {value} from "./data"; export const query = value;');
 const second={...intent,inspectionId:'dependency-edit'};
 expect(await reconciler().inspect(second)).toMatchObject({status:'recorded'});
 const secondRow=await load(second.inspectionId);expect(secondRow.deltaRef?.digest).not.toBe(firstRow.deltaRef?.digest);
 const secondFindings=await Promise.all(secondRow.findingRefs.map(ref=>h.ledger.load(ref)));
 expect(secondFindings).toContainEqual(expect.objectContaining({status:'found',snapshot:expect.objectContaining({finding:expect.objectContaining({category:'dependency',risk:'high',material:true,ambiguous:false,summary:expect.stringContaining('no-query-data')})})}));
 const secondBody=await h.vault.open(secondRow.deltaRef!,{requesterRunRef:claim.runRef});
 if(secondBody.status!=='ready')throw Error('missing delta');expect(JSON.parse(secondBody.record.body).changes.some((c:{level:string})=>c.level==='edge')).toBe(true);
 await expect(reconciler().inspect({...intent,inspectionId:'bad-pin',baselinePin:{...pin,digest:'0'.repeat(64)}})).resolves.toMatchObject({status:'fail_closed',code:'baseline_digest_mismatch'});
 expect(await reconciler().inspect({...intent,inspectionId:'stale',workspaceRevision:2})).toMatchObject({status:'fail_closed',code:'workspace_unavailable'});
 expect(await reconciler().inspect(first)).toMatchObject({status:'fail_closed',code:'recording_rejected'});
 expect(await h.ledger.load(pin.ref)).toMatchObject({status:'found',snapshot:{content:{sourceBinding:source}}});
},60000);
