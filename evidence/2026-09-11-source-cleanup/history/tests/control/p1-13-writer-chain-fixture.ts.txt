 *   - every assertion is a real state assertion (never a mock).

/** 尚未落地的实现以固定文案报告缺口；其它异常必须照常抛出。 */
function isLaneStub(error: unknown): boolean {
  return error instanceof Error && error.message.includes("not implemented yet");
}

export type P113AdvanceResult = { path: "real" | "fixture"; receipt: AdvanceRemediationTaskReceipt };

/**
 * Advance one remediation task step. Real engine first; on the lane-B stub the
 * shared fixture fold commits the SAME state (blindly) so the chain stays
 * runnable on the current baseline. `state` is updated in place so chained
 * steps see the correct aggregate revision.
 */
export async function advanceRemediationStep(
  h: P113WriterHarness,
  state: { task: RemediationTaskV1; revision: number },
  step: { status: "writing" | "verifying" | "resolved"; writerRunRef?: RunRef | null; evidenceRefs?: EvidenceRef[]; result?: RemediationTaskV1["result"] },
  commandId: string,
): Promise<P113AdvanceResult> {
  const cmd = buildP113AdvanceTaskCommand(state.task.taskId, state.revision, {
    status: step.status,
    writerRunRef: step.writerRunRef ?? null,
    evidenceRefs: step.evidenceRefs ?? [],
    result: step.result ?? null,
  }, { commandId });
  const nextTask: RemediationTaskV1 = { ...state.task, status: step.status, writerRunRef: step.writerRunRef ?? null, evidenceRefs: step.evidenceRefs ?? [], result: step.result ?? null, updatedAt: P113_SCHEMA };
  try {
    const r = await h.advanceRemediationTask(cmd);
    state.task = nextTask;
    state.revision = state.revision + 1;
    return { path: "real", receipt: r };
  } catch (e) {
    if (!isLaneStub(e)) throw e;
    const receipt = (await h.ledger.commit(buildP113TaskAdvanceCommit(cmd, { eventId: "evt-p113-wc-adv-" + state.revision, occurredAt: P113_SCHEMA, nextRevision: state.revision + 1, task: nextTask }))) as unknown as AdvanceRemediationTaskReceipt;
    state.task = nextTask;
    state.revision = state.revision + 1;
    return { path: "fixture", receipt };
  }
}

/** Whether the lane-B remediation engine is LIVE (not the baseline stub). */
export async function remediationEngineLive(h: P113WriterHarness): Promise<boolean> {
  const cmd = buildP113AdvanceTaskCommand("task-live-probe", 0, { status: "writing", writerRunRef: null, evidenceRefs: [], result: null }, { commandId: "p113-wc-live-probe" });
  try {
    await h.advanceRemediationTask(cmd);

  } catch (e) {

  }
}

export async function endP113WriterRun(h: P113WriterHarness): Promise<RunRef> {
  const runId = "run-p113-writer";
  const attemptId = "att-p113-writer";
  const claim = await h.claimTask(
    buildDispatchClaimCommand({
      commandId: "cmd-p113-wc-claim",
      correlationId: "corr-p113-wc-claim",
      submittedAt: P113_SCHEMA,
      projectId: P113_PROJECT,
      taskId: P113_TASK,
      goalId: "goal-1",
      runId,
      attemptId,
      roleBinding: P107_ROLE_BINDING_WRITER_V1,
      declaredPermissions: P107_DECLARED_WRITE_PERMISSIONS_V1,
      budget: P107_BUDGET_WRITER_V1,
      idempotencyKey: "p113-wc-claim-idem",
    }),
  );
  expect(claim.status).toBe("committed");
  if (claim.status !== "committed") throw new Error("writer claim failed");

  // Drive the pending outbox intent so the writer run progresses to ENDED
  // (REAL P1-03 path; the default runtime completes the run).
  const drive = await h.drive({ reason: "p113-wc-writer", maxIntents: 4 });
  expect(drive.failures).toHaveLength(0);

  const runLoad = await h.ledger.load(runRefFor(P113_PROJECT, "goal-1", runId));
  expect(runLoad.status).toBe("found");
  if (runLoad.status !== "found") throw new Error("writer run not found");
  const runSnapshot = (runLoad as { snapshot: { status: string } }).snapshot;
  expect(runSnapshot.status).toBe("ended");
  return runRefFor(P113_PROJECT, "goal-1", runId);
}

export type P113WriterChainResult = {
  finding: ArchitectureFindingV1;
  findingSnapshot: string;
  patchReceipt: SubmitRemediationPlanPatchReceipt;
  taskInitial: RemediationTaskV1;
  workspaceRevisionBefore: number;
  workspaceRevisionAfter: number;
  leaseReceipt: AcquireWriteLeaseReceipt;
  recordPatchReceipt: RecordPatchReceipt;
  evidenceRef: EvidenceRef;
  writerRun: RunRef;
  planSnapshot: PlanRevisionSnapshot;
  advanceWriting: P113AdvanceResult;
  advanceVerifying: P113AdvanceResult;
  advanceResolved?: P113AdvanceResult;
  taskResolved?: RemediationTaskV1;
  taskState: { task: RemediationTaskV1; revision: number };
};

export async function runP113WriterChain(
  h: P113WriterHarness,
  opts: { stopAt?: "writing" | "verifying" | "resolved" } = {},
): Promise<P113WriterChainResult> {
  // 1. REAL governance + goal + plan baseline.
  await p111BootstrapGoalGovernance(h.ledger, P113_PROJECT);

  // 2. BEST-EFFORT evolution policy install + activation (lane A not landed).
  //    The real engine throws (stub) AND the shared fixture fold is REJECTED by
  //    the ledger: the third governance kind (ArchitectureEvolutionPolicyRevision)
  //    is not yet accepted by commitGovernanceInstall. The writer chain does NOT
  //    depend on the policy being persisted (the remediation patch/task are
  //    recorded below via the shared fixture folds), so a failure is tolerated
  //    and reported as a lane-A gap.
  await installP113EvolutionPolicy(h).catch(() => undefined);
  await activateP113EvolutionPolicy(h, 1).catch(() => undefined);

  // 3. REAL finding record (findingId "finding-p113-1" — matches the P113 refs).
  const finding: ArchitectureFindingV1 = { ...buildP112DeltaFinding(), findingId: P113_FINDING };
  const findingReceipt = await h.recordArchitectureFinding(buildRecordArchitectureFindingCommand(finding, { commandId: "p113-wc-finding" }));
  expect(findingReceipt.status).toBe("committed");
  const findingSnapshot = await h.ledger.load({ aggregateType: "ArchitectureFinding" as const, projectId: P113_PROJECT, workspaceId: P113_WORKSPACE, findingId: P113_FINDING });
  expect(findingSnapshot.status).toBe("found");
  const findingSnapshotJson = JSON.stringify((findingSnapshot as { snapshot: unknown }).snapshot);

  // 4. REAL plan accept over the P113 goal.
  const planCmd = buildApplyPlanCommand(P113_PLAN_FIXTURE, {
    commandId: "p113-wc-apply-plan",
    correlationId: "corr-p113-wc-apply-plan",
    submittedAt: P113_SCHEMA,
    projectId: P113_PROJECT,
    expectedRevision: 1,
    idempotencyKey: "p113-wc-apply-plan-idem",
  });
  const applyReceipt = await h.applyPlan(planCmd);
  expect(applyReceipt.status).toBe("committed");
  const planLoad = await h.ledger.load(planRevisionRefFor(planCmd));
  expect(planLoad.status).toBe("found");
  const planSnapshot = (planLoad as { snapshot: PlanRevisionSnapshot }).snapshot;

  // 5. REAL-or-fixture remediation plan patch + task record (lane B stub).
  const patch = buildP113PlanPatchV1();
  const patchReceipt = await submitRemediationPlanPatchRealOrFallback(h, patch);
  expect(patchReceipt.receipt.status).toBe("committed");
  const taskCreated = await createRemediationTaskRealOrFallback(h, patch, P113_TASK);
  expect(taskCreated.receipt.status).toBe("committed");

  // 6. REAL exclusive-writer run (P1-03 dispatch) — ENDED run for recordPatch.
  const writerRun = await endP113WriterRun(h);

  // 7. REAL write lease (P1-07), then 8. REAL recordPatch (atomic N -> N+1).
  const workspaceBefore = await h.ledger.load({ aggregateType: "Workspace" as const, projectId: P113_PROJECT, workspaceId: P113_WORKSPACE });
  expect(workspaceBefore.status).toBe("found");
  const workspaceRevisionBefore = (workspaceBefore as { snapshot: { revision: number } }).snapshot.revision;
  const leaseReceipt = await acquireP113WriteLease(h, writerRun);

  const recordPatchReceipt = await h.recordPatch(buildP113RecordPatchCommand(writerRun, planSnapshot, workspaceRevisionBefore, workspaceRevisionBefore + 1));
  expect(recordPatchReceipt.status).toBe("committed");
  const workspaceRevisionAfter = recordPatchReceipt.status === "committed" ? recordPatchReceipt.workspaceRevision : workspaceRevisionBefore + 1;
  expect(workspaceRevisionAfter).toBe(workspaceRevisionBefore + 1);

  // 9. Task: writing -> (REAL evidence at the post-fix revision) -> verifying -> resolved.
  const taskState: { task: RemediationTaskV1; revision: number } = { task: taskCreated.task, revision: 1 };
  const advanceWriting = await advanceRemediationStep(h, taskState, { status: "writing", writerRunRef: writerRun }, "p113-wc-adv-write");
  expect(advanceWriting.receipt.status).toBe("committed");

  const evidenceRef = await submitP113VerificationEvidence(h, planSnapshot, workspaceRevisionAfter, "p113-evidence-writer", writerRun);

  const advanceVerifying = await advanceRemediationStep(h, taskState, { status: "verifying", writerRunRef: writerRun, evidenceRefs: [evidenceRef] }, "p113-wc-adv-verify");
  expect(advanceVerifying.receipt.status).toBe("committed");
  if (opts.stopAt === "verifying") {
    return {
      finding, findingSnapshot: findingSnapshotJson, patchReceipt: patchReceipt.receipt, taskInitial: taskCreated.task,
      workspaceRevisionBefore, workspaceRevisionAfter, leaseReceipt, recordPatchReceipt, evidenceRef, writerRun, planSnapshot,
      advanceWriting, advanceVerifying, taskState,
    };
  }
  const advanceResolved = await advanceRemediationStep(h, taskState, {
    status: "resolved",
    writerRunRef: writerRun,
    evidenceRefs: [evidenceRef],
    result: { workspaceRevisionAfter, verified: true, outcome: "PASS" },
  }, "p113-wc-adv-resolve");
  expect(advanceResolved.receipt.status).toBe("committed");

  const taskLoad = await h.ledger.load(remediationTaskRefFor(P113_PROJECT, P113_WORKSPACE, P113_TASK));
  expect(taskLoad.status).toBe("found");
  const taskResolved = (taskLoad as { snapshot: { task: RemediationTaskV1 } }).snapshot.task;

  return {
    finding, findingSnapshot: findingSnapshotJson, patchReceipt: patchReceipt.receipt, taskInitial: taskCreated.task,
    workspaceRevisionBefore, workspaceRevisionAfter, leaseReceipt, recordPatchReceipt, evidenceRef, writerRun, planSnapshot,
    advanceWriting, advanceVerifying, advanceResolved, taskResolved, taskState,
  };
}
