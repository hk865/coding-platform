export async function sqliteLedger(): Promise<{ ledger: SqliteStateLedger; directory: string; path: string }> {
  const directory = await mkdtemp(path.join(tmpdir(), 'rw04-autonomous-rework-'));
  const file = path.join(directory, 'ledger.sqlite');

export function engineFor(ledger: Ledger): ReturnType<typeof createControlEngine> {
  const deps = createDeterministicDeps();
  return createControlEngine({ ledger, now: deps.clock, eventId: deps.eventId });
}

function scopeFor(projectId: string) {
  const scope = MULTI_SCOPE_CREATE_GOAL_FIXTURE_V1.scopes.find((s) => s.projectId === projectId);
  if (!scope) throw new Error('unknown project: ' + projectId);
  return scope;
}

/**
 * 建立真实前置状态：bootstrap → CreateGoal → 安装并激活两项治理 → 接受手写计划。
 * 直接提交既有 fold 构造器（与 tests/control/goal-change.test.ts 同一手法），
 * 使引擎的 eventId/now 不被前置步骤消费。
 */
export async function prepareScope(ledger: Ledger, projectId: string = RW04_PROJECT): Promise<{ goal: GoalSnapshot; sourcePlan: PlanRevisionSnapshot }> {
  const bootCmd = buildBootstrapCommand(WORKSPACE_BOOTSTRAP_FIXTURE_V1, { commandId: 'cmd-bootstrap', correlationId: 'corr-bootstrap', submittedAt: FIXED });
  const bootReceipt = await ledger.commit(buildBootstrapLedgerCommit(bootCmd, { eventIds: ['evt-bootstrap-1', 'evt-bootstrap-2', 'evt-bootstrap-3', 'evt-bootstrap-4'], occurredAt: FIXED }));
  if (bootReceipt.status !== 'committed') throw new Error('bootstrap failed: ' + JSON.stringify(bootReceipt));

  const scope = scopeFor(projectId);
  const goalReceipt = await ledger.commit(
    buildGoalCreateLedgerCommit(buildCreateGoalCommand(scope, { commandId: 'cmd-goal-' + projectId, correlationId: 'corr-goal-' + projectId, submittedAt: FIXED }), {
      eventId: 'evt-goal-' + projectId,
      occurredAt: FIXED,
      projectRevision: 1,
      workspaceRevision: 1,
    }),
  );
  if (goalReceipt.status !== 'committed') throw new Error('create goal failed: ' + JSON.stringify(goalReceipt));

  const cp = buildInstallCommand(COMPLETION_POLICY_FIXTURE_V1, {
    commandId: 'cmd-install-cp-' + projectId, correlationId: 'corr-install-cp-' + projectId, submittedAt: FIXED, projectId, idempotencyKey: 'inst-cp-' + projectId,
  }) as InstallCompletionPolicyRevisionCommand;
  await ledger.commit(buildInstallLedgerCommit(cp, { eventId: 'evt-install-cp-' + projectId, occurredAt: FIXED }));
  const ab = buildInstallCommand(ARCHITECTURE_BASELINE_FIXTURE_V1, {
    commandId: 'cmd-install-ab-' + projectId, correlationId: 'corr-install-ab-' + projectId, submittedAt: FIXED, projectId, idempotencyKey: 'inst-ab-' + projectId,
  }) as InstallArchitectureBaselineRevisionCommand;
  await ledger.commit(buildInstallLedgerCommit(ab, { eventId: 'evt-install-ab-' + projectId, occurredAt: FIXED }));
  await ledger.commit(
    buildActivateLedgerCommit(
      buildActivateCommand(completionPolicyPinFor(cp), { commandId: 'cmd-act-cp-' + projectId, correlationId: 'corr-act-cp-' + projectId, submittedAt: FIXED, projectId, expectedRevision: 1, idempotencyKey: 'act-cp-' + projectId }),
      { eventId: 'evt-act-cp-' + projectId, occurredAt: FIXED, activeAggregateRevision: 1, projectRevision: 1 },
    ),
  );
  await ledger.commit(
    buildActivateLedgerCommit(
      buildActivateCommand(architectureBaselinePinFor(ab), { commandId: 'cmd-act-ab-' + projectId, correlationId: 'corr-act-ab-' + projectId, submittedAt: FIXED, projectId, expectedRevision: 1, idempotencyKey: 'act-ab-' + projectId }),
      { eventId: 'evt-act-ab-' + projectId, occurredAt: FIXED, activeAggregateRevision: 1, projectRevision: 1 },
    ),
  );

  // RW-07：源计划必须像真实被接受的计划那样携带指派——返工任务的 role 取自被取代任务的指派，
  // 返工草稿的 assignments 也由「源指派 + 增量」推导。其余字段逐字不动。
  const sourceDraft = structuredClone(HAND_AUTHORED_PLAN_REVISION_FIXTURE_V1);
  sourceDraft.assignments = sourceDraft.tasks
    .filter((task) => task.taskKind === 'work')
    .map((task) => ({ taskId: task.taskId, role: 'executor' as const, instruction: '实现并验证「' + task.title + '」：按义务正文产出可独立复核的结果与运行事实。' }));
  const acceptCmd = buildApplyPlanCommand(sourceDraft, {
    commandId: 'cmd-accept-plan-' + projectId, correlationId: 'corr-accept-plan-' + projectId, submittedAt: FIXED, projectId, expectedRevision: 1, idempotencyKey: 'accept-plan-' + projectId,
  });
  const baseGoal = goalSnapshotFor(buildCreateGoalCommand(scope, { commandId: 'cmd-goal-' + projectId, correlationId: 'corr-goal-' + projectId, submittedAt: FIXED }));
  const planReceipt = await ledger.commit(
    buildPlanLedgerCommit(acceptCmd, {
      eventId: 'evt-accept-plan-' + projectId,
      occurredAt: FIXED,
      acceptedAt: FIXED,
      pins: { completionPolicy: completionPolicyPinFor(cp), architectureBaseline: architectureBaselinePinFor(ab) },
      baseGoal,
    }),
  );
  if (planReceipt.status !== 'committed') throw new Error('apply plan failed: ' + JSON.stringify(planReceipt));

  const goal = await loadGoal(ledger, projectId);
  const plan = await loadActivePlan(ledger, goal, projectId);
  return { goal, sourcePlan: plan };
}

export async function loadGoal(ledger: Ledger, projectId: string = RW04_PROJECT, goalId: string = RW04_GOAL): Promise<GoalSnapshot> {
  const loaded = await ledger.load({ aggregateType: 'Goal', projectId, goalId });
  if (loaded.status !== 'found' || loaded.snapshot.ref.aggregateType !== 'Goal') throw new Error('goal not found');
  return loaded.snapshot as GoalSnapshot;
}

export async function loadPlan(ledger: Ledger, planId: string, projectId: string = RW04_PROJECT): Promise<PlanRevisionSnapshot> {
  const loaded = await ledger.load(p111PlanRef(planId, projectId));
  if (loaded.status !== 'found' || loaded.snapshot.ref.aggregateType !== 'PlanRevision') throw new Error('plan not found: ' + planId);
  return loaded.snapshot as PlanRevisionSnapshot;
}

export async function loadActivePlan(ledger: Ledger, goal: GoalSnapshot, projectId: string = RW04_PROJECT): Promise<PlanRevisionSnapshot> {
  if (goal.activePlanRevision === null) throw new Error('goal has no active plan');
  return loadPlan(ledger, goal.activePlanRevision.planId, projectId);
}

/** 安装并激活一份协调策略（真实入口），默认用 P1-15 夹具内容（maxAutonomousReworks = 1）。 */
export async function activateCoordinationPolicy(
  engine: ReturnType<typeof createControlEngine>,
  projectId: string,
  content: CoordinationPolicyContentV1 = P115_COORDINATION_POLICY_CONTENT,
  policyId: string = P115_POLICY,
): Promise<{ policyId: string; content: CoordinationPolicyContentV1; contentDigest: string }> {
  const digest = coordinationPolicyContentDigest(content, policyId, 1);
  const install = buildP115InstallCommand(projectId, { commandId: 'rw04-cmd-install-policy-' + policyId });
  const installReceipt = await engine.installCoordinationPolicy({ ...install, payload: { ...install.payload, policyId, content, contentDigest: digest } });
  if (installReceipt.status !== 'committed') throw new Error('install policy failed: ' + JSON.stringify(installReceipt));
  const activate = buildP115ActivateCommand(projectId, { commandId: 'rw04-cmd-activate-policy-' + policyId, expectedRevision: 1 });
  const activateReceipt = await engine.activateCoordinationPolicy({ ...activate, payload: { target: { ref: { aggregateType: 'CoordinationPolicyRevision', projectId, policyId, revision: 1 }, digest } } });
  if (activateReceipt.status !== 'committed') throw new Error('activate policy failed: ' + JSON.stringify(activateReceipt));
  return { policyId, content, contentDigest: digest };
}

/**
 * 真接纳一条失败结论（工具轮次聚合 Evidence 的形态：kind=observation + coverage + anchor）。
 * 走的是 Control 的既有 submitEvidence 入口，因此边界 (a) 读到的是真实账本事实。
 */
export async function admitConclusion(
  // 只要求 ControlEngine 的证据受理面：harness 与裸引擎都能传入（RW-06 起共用）。
  engine: Pick<import('../../src/contracts/modules.js').ControlEngine, 'submitEvidence'>,
  ledger: Ledger,
  input: {
    plan: PlanRevisionSnapshot;
    taskId: string;
    obligationId: string;
    requirementId: string;
    outcome: EvidenceOutcome;
    evidenceId: string;
    projectId?: string;
    goalId?: string;
    workspaceRevision?: number;
  },
): Promise<void> {
  const projectId = input.projectId ?? RW04_PROJECT;
  const goalId = input.goalId ?? RW04_GOAL;
  const evidence = buildEvidenceV1({
    evidenceId: input.evidenceId,
    kind: 'observation',
    outcome: input.outcome,
    projectId,
    goalId,
    taskId: input.taskId,
    coverage: [{ obligationId: input.obligationId, requirementId: input.requirementId }],
    anchor: {
      schemaVersion: 1,
      planRef: input.plan.ref,
      planRevision: input.plan.planRevision,
      workspaceRevision: input.workspaceRevision ?? 1,
      pinnedCompletionPolicy: { ...input.plan.effectiveCompletionPolicy },
      pinnedArchitectureBaseline: { ...input.plan.effectiveArchitectureBaseline },
    },
    // planDigest 必须是 sha256 hex（账本校验要求）；夹具用计划身份的确定性摘要，不伪造随机值。
    verificationPlanRef: { planId: input.plan.planId, planDigest: sha256Hex('fixture-plan-digest:' + input.plan.planId) },
    runRef: null,
    checkId: input.evidenceId,
    actor: { kind: 'system', id: 'verification-engine' },
    summaryText: '夹具结论 ' + input.obligationId + '/' + input.requirementId + '：' + input.outcome,
    artifactRef: null,
  });
  const receipt = await engine.submitEvidence(
    buildSubmitEvidenceCommand({
      commandId: input.evidenceId,
      idempotencyKey: input.evidenceId,
      correlationId: input.evidenceId + '-corr',
      submittedAt: FIXED,
      actor: { kind: 'system', id: 'verification-engine' },
      evidence,
    }),
  );
  if (receipt.status !== 'committed') throw new Error('evidence admission failed: ' + JSON.stringify(receipt));
  void ledger;
}

/**
 * 构造一条与 VerificationEngine 只读出口一致的问题（身份用权威函数推导）。
 * currentness 标注为 open：本夹具的场景都是「问题相对当前 active revision 仍成立」。
 */
export function buildIssue(input: {
  taskId: string;
  plan: PlanRevisionSnapshot;
  failedRequirements: { obligationId: string; requirementId: string; kind?: 'static' | 'dynamic' | 'reviewer'; result?: 'FAIL' | 'INCONCLUSIVE' | null }[];
  roundId?: string;
  projectId?: string;
  goalId?: string;
  workspaceId?: string;
  summary?: string;
}): ReworkIssueViewV1 {
  const projectId = input.projectId ?? RW04_PROJECT;
  const goalId = input.goalId ?? RW04_GOAL;
  const source = {
    kind: 'verification_round' as const,
    roundId: input.roundId ?? 'round-rw04-1',
    requestId: (input.roundId ?? 'round-rw04-1') + '-request',
    outcome: 'FAIL' as const,
    status: 'completed' as const,
    sourceDigest: 'source-digest-' + input.roundId,
  };
  const failedRequirements = input.failedRequirements.map((requirement) => ({
    obligationId: requirement.obligationId,
    requirementId: requirement.requirementId,
    kind: requirement.kind ?? ('dynamic' as const),
    reason: '夹具失败要求 ' + requirement.obligationId + '/' + requirement.requirementId,
    failure: {
      checkId: 'check-' + requirement.obligationId + '-' + requirement.requirementId,
      failedCheckIds: ['check-' + requirement.obligationId + '-' + requirement.requirementId],
      kind: (requirement.kind ?? 'dynamic') as 'static' | 'dynamic' | 'reviewer',
      category: 'tool_check',
      result: requirement.result === undefined ? ('FAIL' as const) : requirement.result,
      command: 'pnpm test',
      timeoutMs: 1000,
      exitCode: 1,
      timedOut: false,
      durationMs: 10,
      stderrExcerpt: 'fixture failure',
      sourceDigest: source.sourceDigest,
      reportRef: null,
      gaps: [],
    },
  }));
  const issueId = reworkIssueIdFor({ taskId: input.taskId, source, failedRequirements });
  const issue: ReworkIssueV1 = {
    schemaVersion: 1,
    issueId,
    projectId,
    workspaceId: input.workspaceId ?? RW04_WORKSPACE,
    goalId,
    taskId: input.taskId,
    runRef: { aggregateType: 'Run', projectId, goalId, runId: 'run-rw04-1' },
    planRef: input.plan.ref,
    planRevision: input.plan.planRevision,
    source,
    failedRequirements,
    evidenceRefs: [],
    summary: input.summary ?? '夹具问题：任务 ' + input.taskId + ' 在 ' + input.plan.planId + ' 上有未通过要求',
    detectedAt: FIXED,
  };
  return { ...issue, currentness: { status: 'open', issues: [] }, reworkTaskId: reworkTaskIdFor(issueId) };
}

/** 用 RW-03 的真实编译器产出提案；结论不是 proposal 时直接失败（夹具不伪造提案）。 */
export function compileProposal(goal: GoalSnapshot, plan: PlanRevisionSnapshot, issues: ReworkIssueViewV1[], projectId: string = RW04_PROJECT, workIdentities?: PlanningTaskWorkMaterial[]): ReworkProposalV1 {
  const result: ReworkCompileResultV1 = new ReworkPlanCompiler().compile({
    schemaVersion: 1,
    projectId,
    workspaceId: goal.workspaceRef.workspaceId,
    goalRef: goal.ref,
    goalObjective: goal.objective,
    activePlan: plan,
    ...(workIdentities ? { workIdentities } : {}),
    issues,
  });
  if (result.status !== 'proposal') throw new Error('rework compiler did not produce a proposal: ' + JSON.stringify(result));
  return result.proposal;
}
