/**
 * RW-16：夹具允许把该 Run 的角色绑定换成某个**已登记角色**（默认仍是 P1-07 的短生命周期绑定）。
 * 判据没有改变：角色矩阵生效后，claim 守卫按 RW-11 的同一份策略判定声明权限与 revision。
 */
type FixtureOptions = {
  roleBinding?: typeof P107_ROLE_BINDING_WRITER_V1;
  declaredPermissions?: { tools: string[]; writeScope: string[] };
};

async function fixture(kinds: Array<'static' | 'dynamic' | 'reviewer'> = ['dynamic', 'dynamic', 'static', 'reviewer'], options: FixtureOptions = {}) {
  const root = await mkdtemp(join(tmpdir(), 'verification-round-')); roots.push(root);
  const directory = await mkdtemp(join(tmpdir(), 'verification-round-journal-')); roots.push(directory);
  await writeFile(join(root, 'source.txt'), 'source');
  const h = createInMemoryHarness({ deps: { clock: () => P107_SCHEMA } }), p107 = toP1_07Harness(h);
  const draft = structuredClone(P107_PLAN_REVISION_FIXTURE_V1);
  draft.obligations.push({ obligationId: 'round-obligation', title: 'Explicit tool and Reviewer requirements', requirementLevel: 'required', taskIds: [scope.taskId],
    verificationRequirements: kinds.map((kind, index) => ({ requirementId: 'round-vr-' + index, kind, requirementLevel: 'required', description: 'required ' + kind })) });
  await prepareP107Scenario(p107, draft);
  await runP107Task(p107, { taskId: scope.taskId, runId: scope.runId, attemptId: 'round-attempt', roleBinding: options.roleBinding ?? P107_ROLE_BINDING_WRITER_V1, declaredPermissions: options.declaredPermissions ?? { tools: ['read', 'write'], writeScope: ['*'] }, budget: P107_BUDGET_WRITER_V1 });
  const context = new VerificationContextCompiler({ ledger: h.ledger, vault: h.vault, runtime: { all: () => [{ spec: scope, status: 'completed' }] }, rootFor: () => root, workspaceSource: new CandidateWorkspaceReader(), roundSource: new VerificationWorkspaceReader() });
  const deps: VerificationServiceDeps = { directory, disposition: new ControlReworkDisposition(h.ledger as never), context, vault: h.vault, control: h.control, workspaceLease: h.workspaceLease, candidatePatchCheck: { check: async () => { throw Error('candidate patch checks are not part of round execution'); } } };
  const reopen = async () => { const service = new VerificationService(deps); await service.init(); return service; };
  return { root, directory, h, deps, context, service: await reopen(), reopen };
}

/** 某个已登记角色的绑定（revision 用角色规格 revision 的十进制写法，与矩阵 pin 一致）。 */
function roleBindingFor(roleId: string): RoleBindingRefV1 {
  return { schemaVersion: 1, bindingId: 'binding-' + roleId + '-round-v1', templateId: roleId, templateRevision: String(ROLE_SPEC_REVISION), bindingVersion: 1, policyRevision: 'auth-policy-runtime-v1' };
}

/**
 * 用**产品自己的治理命令**安装并激活角色规格 + 一份含 roles 的协调策略。
 * 为什么放在轮次用例里：RW-16 要证明的正是「装了 executor／integrator 规格之后轮次仍然能完成」，
 * 所以安装必须走 install+activate 三段式（CAS 与摘要都由 ControlEngine 判定），不能替身伪造。
 */
async function installRole(s: Awaited<ReturnType<typeof fixture>>, roleId: string): Promise<void> {
  const h = s.h;
  const identity = (id: string) => ({ projectId: P107_PROJECT, actor: { kind: 'human' as const, id: 'user-1' }, idempotencyKey: id, commandId: id, correlationId: id, submittedAt: P107_SCHEMA });
  const projectRevision = async (): Promise<number> => {
    const loaded = await h.ledger.load({ aggregateType: 'Project', projectId: P107_PROJECT } as never);
    if (loaded.status !== 'found') throw Error('project missing');
    return loaded.snapshot.revision;
  };
  expect((await h.control.installRoleSpec(buildRoleSpecInstallCommandFor(roleId, identity('inst-' + roleId)))).status).toBe('committed');
  expect((await h.control.activateRoleSpec(buildRoleSpecActivateCommandFor(roleId, { ...identity('act-' + roleId), expectedRevision: await projectRevision() }))).status).toBe('committed');
  const policyId = 'coordination-' + roleId;
  const content = buildCoordinationPolicyContentWithRolesV1(P107_PROJECT, [roleId]);
  expect((await h.control.installCoordinationPolicy(buildCoordinationPolicyInstallCommand({ policyId, content }, identity('inst-policy-' + roleId)))).status).toBe('committed');
  const pin = {
    ref: { aggregateType: 'CoordinationPolicyRevision' as const, projectId: P107_PROJECT, policyId, revision: P15_COORDINATION_POLICY_REVISION },
    digest: roleMatrixPolicyDigest(content, policyId),
  };
  expect((await h.control.activateCoordinationPolicy(buildCoordinationPolicyActivateCommand(pin, { ...identity('act-policy-' + roleId), expectedRevision: await projectRevision() }))).status).toBe('committed');
  // 轮次侧消费角色规格的判据与派发侧是**同一个**端口实现（ControlEngine 的策略），这里原样注入。
  s.deps.roleSpec = new LedgerRoleSpecRead({ ledger: h.ledger });
}

/** 本 Run 当前的冻结材料（用于取得 canonical run 与工作区版本）。 */
async function roundRun(s: Awaited<ReturnType<typeof fixture>>): Promise<{ run: RunSnapshot; workspaceRevision: number }> {
  const material = await s.context.resolveRound(scope);
  if (material.status !== 'ready') throw Error('fixture material unavailable: ' + JSON.stringify(material));
  return { run: material.material.run, workspaceRevision: material.material.workspaceRevision };
}

/** 真实 recordPatch：本 Run 的实现结果（先取活动写租约，再原子登记并推进工作区版本）。 */
async function recordRunPatch(s: Awaited<ReturnType<typeof fixture>>, patchId: string): Promise<void> {
  const { run, workspaceRevision } = await roundRun(s);
  const leaseId = 'round-patch-lease-' + patchId;
  const attemptRef = taskAttemptRefFor(P107_PROJECT, P107_GOAL, run.task.taskId, run.attemptId);
  expect((await s.h.workspaceLease.acquireWriteLease(buildAcquireWriteLeaseCommand({
    commandId: leaseId, leaseId, idempotencyKey: leaseId, correlationId: leaseId, submittedAt: P107_SCHEMA,
    actor: { kind: 'system', id: 'round-patch' }, projectId: P107_PROJECT, workspaceId: P107_WORKSPACE,
    scope: P107_SCOPE_WRITER, declaredWriteScope: P107_DECLARED_WRITE_PERMISSIONS_V1.writeScope,
    holder: { runRef: run.ref, attemptRef, roleBinding: run.roleBinding }, expiresAt: null,
  })))).toMatchObject({ status: 'committed' });
  const patch: PatchArtifactV1 = {
    schemaVersion: 1, patchId, projectId: P107_PROJECT, workspaceId: P107_WORKSPACE, goalId: P107_GOAL,
    taskId: run.task.taskId, planRef: run.planRef, taskRevision: 1, runRef: run.ref, attemptRef,
    roleBinding: run.roleBinding, kind: 'patch', title: '本 Run 的实现结果', changedPaths: ['src/p107/round-fix.ts'],
    bodyRef: buildP107ArtifactRef(patchId), beforeWorkspaceRevision: workspaceRevision, afterWorkspaceRevision: workspaceRevision + 1,
    checkResults: [{ checkId: 'round-patch-check', outcome: 'PASS', summary: 'deterministic checks passed' }],
    usedInputEvidenceRefs: [], generatedAt: P107_SCHEMA,
  };
  expect((await s.h.recordPatch(buildP107RecordPatchCommand({ commandId: 'cmd-' + patchId, projectId: P107_PROJECT, patch }))).status).toBe('committed');
}

describe('durable explicit verification rounds', () => {
  it('runs every applicable immutable check once, merges repeated coverage, aggregates FAIL over later PASS and leaves Reviewer pending', async () => {
    const s = await fixture();
    const input = request(definition('a-fail', 'dynamic', countCommand('a', 1)), definition('b-pass', 'dynamic', countCommand('b')), definition('c-static', 'static', countCommand('c')), { ...definition('foreign'), appliesTo: { workspaceId: 'foreign', taskIds: 'all' } });
    const { round } = await s.service.startRound(scope, input);
    expect(round.status, JSON.stringify(round.gaps)).toBe('completed');
    expect(round.current).toEqual({ status: 'current', issues: [] });
    expect(round.checks).toHaveLength(3);
    expect(round.checks[0]!.coverage).toHaveLength(2);
    expect(round.coverage.map(c => c.result)).toEqual(['FAIL', 'FAIL', 'PASS', null]);
    expect(round.outcome).toBe('FAIL');
    expect(round.gaps.some(g => g.code === 'reviewer_required')).toBe(true);
    expect(round.aggregate!.admissions.map(a => a.coverage.result)).toEqual(['FAIL', 'FAIL', 'PASS']);
    expect(round.aggregate!.admissions.every(a => a.status === 'admitted')).toBe(true);
    for (const id of ['a', 'b', 'c']) expect(await readFile(join(s.root, '.cache', id), 'utf8')).toBe('x');
    const raw = await s.service.checkReports(scope, round.checks[0]!.requestId);
    expect(raw.reports[0]).toMatchObject({ definition: { checkId: 'a-fail' }, category: 'tool_check', result: 'FAIL' });
    await expect(s.service.admitCheckEvidence(scope, { requestId: round.checks[1]!.requestId, reportDigest: round.checks[1]!.record!.progress!.phase === 'report_stored' ? round.checks[1]!.record!.progress!.artifactRef.digest : '' })).rejects.toThrow('完整覆盖聚合');
    const replay = await (await s.reopen()).startRound(scope, input);
    expect(replay.replayed).toBe(true); expect(replay.round.roundId).toBe(round.roundId);
    for (const id of ['a', 'b', 'c']) expect(await readFile(join(s.root, '.cache', id), 'utf8')).toBe('x');
    await expect(s.service.startRound(scope, request(definition('different')))).rejects.toThrow('内容已改变');
    expect(await s.service.roundReceipt(scope, input.requestId)).toMatchObject({ runId: scope.runId, taskId: scope.taskId, roundId: round.roundId });
  });

  it('reopens a body-first partial round using its original full plan and continues only the never-started child', async () => {
    const s = await fixture(['dynamic', 'static']);
    const save = VerificationJournal.prototype.save;
    let stop = true;
    vi.spyOn(VerificationJournal.prototype, 'save').mockImplementation(async function (this: VerificationJournal, kind, id, value) {
      if (stop && kind === 'round' && (value as any).status === 'running' && (value as any).coverage.some((c: any) => c.result !== null)) { stop = false; throw Error('injected stop after first durable report'); }
      return save.call(this, kind, id, value);
    });
    const input = request(definition('a-dynamic', 'dynamic', countCommand('a')), definition('b-static', 'static', countCommand('b')));
    const first = await s.service.startRound(scope, input);
    expect(first.round.status).toBe('interrupted');
    expect(first.round.checks[1]!.record).toBeNull();
    const item = first.round.checks[0]!, key = sha256Hex(canonicalJson([{ projectId: scope.projectId, workspaceId: scope.workspaceId, goalId: scope.goalId, runId: scope.runId }, item.requestId]));
    // The provider report is durable before the derived VerificationResult. A
    // process loss at that exact boundary leaves this supported disk shape.
    const filename = join(s.directory, 'check-' + key + '.json');
    const stored = JSON.parse(await readFile(filename, 'utf8')) as CommandCheckRecord;
    stored.result = null; stored.status = 'running'; stored.lifecycle = 'report_stored';

    expect(await readFile(join(s.root, '.cache/a'), 'utf8')).toBe('x');

    expect(resumed.round.status, JSON.stringify(resumed.round.gaps)).toBe('completed');
    expect(resumed.round.plan!.planDigest).toBe(first.round.plan!.planDigest);

    expect(await readFile(join(s.root, '.cache/original'), 'utf8')).toBe('x');

  it('uses the generic VerificationPort for a real explicit round and serializes concurrent same-key submissions', async () => {
