 * RW-16：VerificationEngine 的**必产出见证通道**（ADR 0003 D4-3）。
 *
 * RW-15 接上了「角色必产出」的消费者，但见证表里七类只有一类（独立审阅原报告）有真实通道，
 * 其余六类 `channel: null` —— 于是只要项目安装 executor 规格，此后每个普通任务的轮次都不再 PASS、
 * Task 永远无法归约成 satisfied。本票为每一类接上**可归因到本次运行的既有已提交事实**，
 * 并且**不放松 fail-closed**：
 *
 *   1. 见证表全键穷尽（`Record<RoleOutputKindV1, ...>`）：新增一类必产出而表里没有键即编译失败；
 *   2. 每条通道读的都是 canonical 聚合或带 runRef 的落账事件，并逐字段复核归属（runRef／taskId／
 *      workspaceId／planRef 全等）；别的运行的事实、模型自述、调用方声明都不算见证；
 *   3. 事实不存在、读不到、账本扫描超限 → 仍是缺项（这里有正反两面的真实用例）。
 *
 * 事实由**真实命令**产生：InMemoryHarness 的真实 ControlEngine／真实账本／真实治理
 * install+activate（RoleSpecRevision + 带 roles 的 CoordinationPolicy）+ 真实 recordPatch／

it('见证表全键穷尽：新增一类必产出必须显式表态由哪条既有事实见证', async () => {
  // 编译期保证：ROLE_OUTPUT_WITNESS_V1 的类型是 Record<RoleOutputKindV1, ...>。在
  // src/contracts/role-spec.ts 的 ROLE_OUTPUT_KINDS 里临时加一个成员，`npx tsc --noEmit -p tsconfig.json`
  // 就会报 "Property '<新成员>' is missing"；删掉该成员即恢复。下面这条赋值在编译期复核同一件事。
  const table: Record<RoleOutputKindV1, { channel: string | null; because: string }> = ROLE_OUTPUT_WITNESS_V1;
  expect(Object.keys(table).sort()).toEqual([...ROLE_OUTPUT_KINDS].sort());
  for (const kind of ROLE_OUTPUT_KINDS) expect(table[kind].because.length).toBeGreaterThan(0);
});

it('RW-16 接通的五类通道逐类点名；仍无通道的两类写明需要的通道与产品后果', async () => {
  const channels: Record<RoleOutputKindV1, RoleOutputWitnessChannelV1> = {
    'answer-with-sources': null,
    proposal: null,
    'conflict-report': 'run-bound-conflict-surface',
    'verification-verdict': 'run-bound-review-output',
    'implementation-result': 'run-bound-patch-record',
    'integration-result': 'run-bound-integration-result',
    'work-record': 'run-bound-execution-note',
  };
  for (const kind of ROLE_OUTPUT_KINDS) expect(ROLE_OUTPUT_WITNESS_V1[kind].channel).toBe(channels[kind]);
  // 没有通道的两类必须把「需要哪条通道」与「产品后果」说出来，不能只写一句「没有通道」。
  for (const kind of ['answer-with-sources', 'proposal'] as const) {
    expect(ROLE_OUTPUT_WITNESS_V1[kind].because).toContain('需要的通道');
    expect(ROLE_OUTPUT_WITNESS_V1[kind].because).toContain('产品后果');
  }
});

it('没有规格可依时如实记 absent：不据此阻断，也不假装核对过', async () => {
  const without = await evaluateRoleOutputCompleteness({ projectId: PROJECT, run: runSnapshot(), materials: outputMaterials(emptyLedger) });
  expect(without.status).toBe('absent');
  expect(without.detail).toContain('RoleSpecReadPort');
  const absent = await evaluateRoleOutputCompleteness({
    projectId: PROJECT, run: runSnapshot(), materials: outputMaterials(emptyLedger),
    roleSpec: { resolve: async () => ({ status: 'absent', roleId: 'executor', reason: '本次运行没有角色规格可依' }) },
  });
  expect(absent.status).toBe('absent');
  const inadmissible = await evaluateRoleOutputCompleteness({
    projectId: PROJECT, run: runSnapshot(), materials: outputMaterials(emptyLedger),
    roleSpec: { resolve: async () => ({ status: 'inadmissible', roleId: 'executor', reasons: [{ code: 'role_binding_not_admissible', roleId: 'executor', detail: 'role_spec_stale', message: '角色已撤下' }] }) },
  });
  expect(inadmissible.status).toBe('absent');
  expect(inadmissible.detail).toContain('角色已撤下');
  const noEnvelope = await evaluateRoleOutputCompleteness({
    projectId: PROJECT, run: runSnapshot({ envelope: null }), materials: outputMaterials(emptyLedger),
    roleSpec: rolePort(specContent([{ kind: 'implementation-result', reason: 'r' }])),
  });
  expect(noEnvelope.status).toBe('absent');
});

type RoleWorld = {

/** 真实治理：按既有 install/activate 三段式安装并激活一份角色规格 + 带 roles 的协调策略。 */
async function installRole(world: RoleWorld, roleId: string): Promise<void> {
  const control = createControlEngine({ ledger: world.h.ledger, now: () => P107_SCHEMA, eventId: deterministic.eventId });
  const identity = (id: string) => ({ projectId: P107_PROJECT, actor: { kind: 'human' as const, id: 'user-1' }, idempotencyKey: id, commandId: id, correlationId: id, submittedAt: P107_SCHEMA });
  const projectRevision = async (): Promise<number> => {
    const loaded = await world.h.ledger.load({ aggregateType: 'Project', projectId: P107_PROJECT } as never);
    if (loaded.status !== 'found') throw Error('project missing');
    return loaded.snapshot.revision;
  };
  expect((await control.installRoleSpec(buildRoleSpecInstallCommandFor(roleId, identity('inst-' + roleId)))).status).toBe('committed');
  expect((await control.activateRoleSpec(buildRoleSpecActivateCommandFor(roleId, { ...identity('act-' + roleId), expectedRevision: await projectRevision() }))).status).toBe('committed');
  const policyId = 'coordination-' + roleId;
  const content = buildCoordinationPolicyContentWithRolesV1(P107_PROJECT, [roleId]);
  expect((await control.installCoordinationPolicy(buildCoordinationPolicyInstallCommand({ policyId, content }, identity('inst-policy-' + roleId)))).status).toBe('committed');
  const pin = { ref: { aggregateType: 'CoordinationPolicyRevision' as const, projectId: P107_PROJECT, policyId, revision: P15_COORDINATION_POLICY_REVISION }, digest: roleMatrixPolicyDigest(content, policyId) };
  expect((await control.activateCoordinationPolicy(buildCoordinationPolicyActivateCommand(pin, { ...identity('act-policy-' + roleId), expectedRevision: await projectRevision() }))).status).toBe('committed');
}

/** 真实 P1-07 场景（真实 bootstrap／治理／目标／计划），但 DAG 边留空：本文件只关心"某个 Run 的产出"。 */
async function world(): Promise<RoleWorld> {
  const h = createInMemoryHarness({ deps: { clock: () => P107_SCHEMA } });
  const p107 = toP1_07Harness(h);
  const draft = structuredClone(P107_PLAN_REVISION_FIXTURE_V1);
  draft.executionDag = { dependsOn: [] };
  const preview = await prepareP107Scenario(p107, draft);
  return { h, p107, workspaceRevision: preview.workspaceRevision };
}

function bindingFor(roleId: string): RoleBindingRefV1 {
  return { schemaVersion: 1, bindingId: 'binding-' + roleId + '-v1', templateId: roleId, templateRevision: String(ROLE_SPEC_REVISION), bindingVersion: 1, policyRevision: 'auth-policy-runtime-v1' };
}

async function runFor(world: RoleWorld, deps: { taskId: string; runId: string; roleId: string }): Promise<RunSnapshot> {
  const roleBinding = bindingFor(deps.roleId);
  const declaredPermissions = deps.roleId === ROLE_SOURCE_EXECUTOR ? P107_DECLARED_WRITE_PERMISSIONS_V1 : { tools: ['read'], writeScope: [] as string[] };
  const runRef = await runP107Task(world.p107, {
    taskId: deps.taskId, runId: deps.runId, attemptId: 'attempt-' + deps.runId,
    roleBinding, declaredPermissions, budget: P107_BUDGET_WRITER_V1,
  });
  const loaded = await world.h.ledger.load(runRef);
  if (loaded.status !== 'found') throw Error('run missing');
  return loaded.snapshot as RunSnapshot;
}

/** 真实 recordPatch：先取活动写租约（holder=本 Run），再登记补丁（原子推进工作区版本）。 */
async function recordRunPatch(world: RoleWorld, run: RunSnapshot, patchId: string): Promise<void> {
  const roleBinding = bindingFor(ROLE_SOURCE_EXECUTOR);
  const attemptRef = taskAttemptRefFor(P107_PROJECT, P107_GOAL, run.task.taskId, run.attemptId);
  const leaseId = 'lease-' + run.ref.runId;
  expect((await world.h.acquireWorkspaceWriteLease(buildP107AcquireWriteLeaseCommand({
    commandId: 'cmd-' + leaseId, projectId: P107_PROJECT, leaseId, scope: P107_SCOPE_WRITER,
    declaredWriteScope: P107_DECLARED_WRITE_PERMISSIONS_V1.writeScope,
    holder: { runRef: run.ref, attemptRef, roleBinding },
  }))).status).toBe('committed');
  const patch: PatchArtifactV1 = {
    schemaVersion: 1, patchId, projectId: P107_PROJECT, workspaceId: P107_WORKSPACE, goalId: P107_GOAL,
    taskId: run.task.taskId, planRef: run.planRef, taskRevision: 1, runRef: run.ref, attemptRef, roleBinding,
    kind: 'patch', title: '本 Run 的实现结果', changedPaths: ['src/p107/fix-a.ts'], bodyRef: buildP107ArtifactRef(patchId),
    beforeWorkspaceRevision: world.workspaceRevision, afterWorkspaceRevision: world.workspaceRevision + 1,
    checkResults: [{ checkId: 'p107-check', outcome: 'PASS', summary: 'deterministic checks passed' }],
    usedInputEvidenceRefs: [], generatedAt: P107_SCHEMA,
  };
  expect((await world.h.recordPatch(buildP107RecordPatchCommand({ commandId: 'cmd-' + patchId, projectId: P107_PROJECT, patch }))).status).toBe('committed');
}

/** 真实 recordIntegrationResult：join 记录带 runRef，聚合按 (project, goal, task) 累积。 */
async function recordRunIntegration(world: RoleWorld, run: RunSnapshot, resultId: string): Promise<void> {
  const result: IntegrationTaskResultV1 = {
    schemaVersion: 1, resultId, projectId: P107_PROJECT, workspaceId: P107_WORKSPACE, goalId: P107_GOAL,
    taskId: run.task.taskId, planRef: run.planRef, taskRevision: 1, runRef: run.ref,
    attemptRef: taskAttemptRefFor(P107_PROJECT, P107_GOAL, run.task.taskId, run.attemptId),
    workspaceRevision: world.workspaceRevision, inputs: [], conflicts: [], gaps: [], explanation: null, escalate: false, generatedAt: P107_SCHEMA,
  };
  expect((await world.h.recordIntegrationResult(buildP107RecordIntegrationCommand({ commandId: 'cmd-' + resultId, projectId: P107_PROJECT, expectedRevision: 0, result }))).status).toBe('committed');

/**
 * 真实 recordExecutionNote：工作身份 + 留痕（body-first，作者运行必须已 link）。
 *
 * RC-03：这个任务的持久工作身份已由派发面在建立 Run 时创建，并且它是**唯一**的那条。
 * 这里从权威解析面取它（不在测试里复制推导规则），而不是用 'work-'+taskId 另造一条 ——
 * 命令面会以 already_bound 拒绝，账本里也不会再有第二条身份。
 */
async function recordRunNote(world: RoleWorld, run: RunSnapshot, noteId: string): Promise<void> {
  const roleBinding = run.roleBinding;
  const resolution = await world.h.control.resolveTaskWorkIdentity({
    projectId: P107_PROJECT, workspaceId: P107_WORKSPACE, goalId: P107_GOAL, taskId: run.task.taskId,
  });
  if (resolution.status !== 'resolved') throw Error('本 Run 的工作身份不可用：' + resolution.status);
  const workId = resolution.binding.workId;
  const note = buildExecutionNoteV1({
    noteId, workId, projectId: P107_PROJECT, workspaceId: P107_WORKSPACE, runRef: run.ref,
    attemptRef: taskAttemptRefFor(P107_PROJECT, P107_GOAL, run.task.taskId, run.attemptId),
    roleBindingRef: roleBinding, kind: 'result',
    applicableVersions: { planRef: run.planRef, planRevision: 1, workspaceRevision: world.workspaceRevision, governanceRevision: null },
    bodyRef: buildP107ArtifactRef(noteId),
  });
  expect((await world.h.recordExecutionNote(buildRecordExecutionNoteCommand({ commandId: 'cmd-note-' + noteId, projectId: P107_PROJECT, note }))).status).toBe('committed');
}
