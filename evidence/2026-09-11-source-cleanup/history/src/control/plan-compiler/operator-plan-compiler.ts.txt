import { createHash, randomUUID } from 'node:crypto';
import { mkdir, readdir, readFile } from 'node:fs/promises';
import { join } from 'node:path';
import type { ControlEngine } from '../../contracts/modules.js';
import type { OperatorPlanningPort, OperatorPlanningContextPort } from '../../contracts/operator-planning.js';
import type { ExplorationScope as Scope, ExplorationPlan } from '../../contracts/exploration.js';
import type { PlanRevisionDraft } from '../../contracts/plan.js';
import { canonicalJson } from '../../contracts/fingerprint.js';
import { buildApplyPlanCommand } from '../../contracts/commands/plan.js';
import { writeAtomicFile } from '../../storage/atomic-file.js';
const digest = (value: string) => createHash('sha256').update(value).digest('hex');
const scopeKey = (scope: Scope) => canonicalJson([scope.projectId, scope.workspaceId, scope.goalId]);
function ensure(value: unknown, message: string): asserts value { if (!value)
    throw Error(message); }
function text(value: unknown, label: string, cap = 4096): string { ensure(typeof value === 'string' && value.trim() && value.length <= cap, '无效字段：' + label); return value.trim(); }
function id(value: unknown, label: string): string { const s = text(value, label, 64); ensure(/^[a-zA-Z0-9-]+$/.test(s), '无效标识：' + label); return s; }
function object(value: unknown): Record<string, unknown> { ensure(value && typeof value === 'object' && !Array.isArray(value), '需要探索任务对象'); return value as Record<string, unknown>; }
function accepted<T extends {
    status: string;
}>(value: T, label = '计划被拒绝'): T { ensure(value.status !== 'rejected', label + '：' + JSON.stringify(value)); return value; }
export class OperatorPlanCompiler implements OperatorPlanningPort {
    private plans: ExplorationPlan[] = [];
    constructor(private readonly deps: {
        directory: string;
        context: OperatorPlanningContextPort;
        control: Pick<ControlEngine, 'applyPlan'>;
        now: () => string;
    }) { }
    async init() {
        await mkdir(this.deps.directory, { recursive: true, mode: 0o700 });
        for (const name of (await readdir(this.deps.directory)).sort()) {
            if (/^plan-[a-f0-9]{64}\.json$/.test(name))
                this.plans.push(JSON.parse(await readFile(join(this.deps.directory, name), 'utf8')) as ExplorationPlan);
        }
        for (const plan of this.plans)
            if (plan.status === 'pending') {
                accepted(await this.deps.control.applyPlan(plan.command), '探索计划恢复失败');
                plan.status = 'accepted';
                await this.save('plan', digest(scopeKey(plan)), plan);
            }
    }
    private save(kind: string, key: string, body: unknown) { return writeAtomicFile(join(this.deps.directory, kind + '-' + key + '.json'), JSON.stringify(body)); }
    plan(scope: Scope) { const plan = this.plans.find(p => scopeKey(p) === scopeKey(scope)); ensure(plan && plan.status === 'accepted', '该目标没有已接受的探索计划'); return structuredClone(plan); }
    acceptedPlans() { return structuredClone(this.plans.filter(plan => plan.status === 'accepted')); }
    async exploration(scope: Scope, input: Record<string, unknown>) {
        const requestId = id(input['requestId'], 'requestId');
        ensure(Array.isArray(input['tasks']) && input['tasks'].length >= 2 && input['tasks'].length <= 8, '探索计划需要 2 到 8 个真实工作节点');
        const tasks = input['tasks'].map(value => { const t = object(value); ensure(Array.isArray(t['dependsOn']) && t['dependsOn'].length <= 7, '依赖必须为任务标识数组'); return { taskId: id(t['taskId'], 'taskId'), title: text(t['title'], 'title', 160), instruction: text(t['instruction'], 'instruction'), dependsOn: t['dependsOn'].map(v => id(v, 'dependsOn')) }; });
        const ids = new Set(tasks.map(t => t.taskId));
        ensure(ids.size === tasks.length && !ids.has('gate-goal'), '任务标识重复或占用验收节点');
        for (const task of tasks)
            ensure(new Set(task.dependsOn).size === task.dependsOn.length && task.dependsOn.every(dep => ids.has(dep) && dep !== task.taskId), '任务依赖无效');
        const visited = new Set<string>(), active = new Set<string>();
        const visit = (taskId: string) => { ensure(!active.has(taskId), '探索依赖不能形成环'); if (visited.has(taskId))
            return; active.add(taskId); tasks.find(t => t.taskId === taskId)!.dependsOn.forEach(visit); active.delete(taskId); visited.add(taskId); };
        tasks.forEach(t => visit(t.taskId));
        const fingerprint = digest(canonicalJson({ scope, requestId, tasks })), prior = this.plans.find(p => scopeKey(p) === scopeKey(scope));
        if (prior) {
            ensure(prior.fingerprint === fingerprint, '该目标已有不同探索计划，请新建目标');
            return { ...this.publicPlan(prior), replayed: true };
        }
        const goal = await this.deps.context.goal(scope);
        ensure(goal.workspaceRef.workspaceId === scope.workspaceId && !goal.activePlanRevision, '请在没有计划的新目标中配置探索');
        const sourceDigest = await this.deps.context.sourceDigest(scope), createdAt = this.deps.now(), planId = 'real-exploration-' + scope.goalId;
        const command = buildApplyPlanCommand({ schemaVersion: 1, planId, planRevision: 1, goalId: scope.goalId, stages: [{ stageId: 'exploration', title: '项目只读探索' }], tasks: [...tasks.map(t => ({ taskId: t.taskId, title: t.title, stageId: 'exploration', taskKind: 'work' as const, requirementLevel: 'required' as const, disposition: 'active' as const, phase: 'pending' as const, scope: { kind: 'stage' as const, stageId: 'exploration' } })), { taskId: 'gate-goal', title: '探索报告整体核验', taskKind: 'gate', requirementLevel: 'required', disposition: 'active', phase: 'pending', scope: { kind: 'goal' } }], obligations: [...tasks.map(t => ({ obligationId: 'explore-' + t.taskId, title: t.title + '：来源核对与操作者审阅', requirementLevel: 'required' as const, taskIds: [t.taskId], verificationRequirements: [{ requirementId: 'source-review-' + t.taskId, requirementLevel: 'required' as const, kind: 'static', description: '完成真实只读探索；由操作者核对报告语义及来源，不以运行结束自动通过' }] })), { obligationId: 'exploration-gate', title: '全部探索报告齐备且经操作者整体核对', requirementLevel: 'required', taskIds: ['gate-goal'], verificationRequirements: [{ requirementId: 'gate-review', requirementLevel: 'required', kind: 'static', description: '所有工作节点正式满足；操作者整体复核；不代表实际构建或运行通过' }] }], assignments: tasks.map(t => ({ taskId: t.taskId, role: 'executor' as const, instruction: t.instruction })), taskHierarchy: { parentOf: tasks.map(t => ({ parentTaskId: 'gate-goal', childTaskId: t.taskId })) }, executionDag: { dependsOn: [...tasks.flatMap(t => t.dependsOn.map(dep => ({ taskId: t.taskId, dependsOnId: dep, requires: { kind: 'artifact' as const, label: '已核对的探索报告：' + dep } }))), ...tasks.map(t => ({ taskId: 'gate-goal', dependsOnId: t.taskId, requires: { kind: 'gate-result' as const, label: '已正式归约的探索结果：' + t.taskId } }))] } }, { projectId: scope.projectId, goalId: scope.goalId, actor: { kind: 'human', id: 'user-1' }, commandId: 'exploration-plan-' + requestId, correlationId: requestId, idempotencyKey: 'exploration-plan-' + requestId, expectedRevision: goal.revision, submittedAt: createdAt });
        const manifest: ExplorationPlan = { ...scope, planOrigin: 'operator', planId, requestId, tasks, gateTaskId: 'gate-goal', sourceDigest, createdAt, fingerprint, command, status: 'pending' };
        await this.save('plan', digest(scopeKey(scope)), manifest);
        this.plans.push(manifest);
        accepted(await this.deps.control.applyPlan(command), '探索计划被拒绝');
        manifest.status = 'accepted';
        await this.save('plan', digest(scopeKey(scope)), manifest);
        return { ...this.publicPlan(manifest), replayed: false };
    }
    private publicPlan(plan: ExplorationPlan) { return { planOrigin: plan.planOrigin, planId: plan.planId, tasks: plan.tasks, gateTaskId: plan.gateTaskId, sourceDigest: plan.sourceDigest, createdAt: plan.createdAt }; }
    async ensureTaskPlan(scope: Scope, instruction: string) {
        const goal = await this.deps.context.goal(scope), goalId = scope.goalId, taskId = 'coding-task';
        if (!goal.activePlanRevision) {
            // New ordinary development plans require independent semantic review.
            // An already accepted plan keeps its original immutable obligations.
            const draft: PlanRevisionDraft = {
                reviewAdmissionProtocol: 'independent-review-v1',
                schemaVersion: 1, planId: `real-plan-${goalId}`, planRevision: 1, goalId,
                stages: [{ stageId: 'coding', title: '真实开发任务' }],
                tasks: [
                    { taskId, stageId: 'coding', title: instruction.slice(0, 160), requirementLevel: 'required', taskKind: 'work', disposition: 'active', phase: 'pending', scope: { kind: 'stage', stageId: 'coding' } },
                    { taskId: 'gate-goal', title: '独立验收', requirementLevel: 'required', taskKind: 'gate', disposition: 'active', phase: 'pending', scope: { kind: 'goal' } },
                ],
                // RW-07：任务与它的指派同属一个 revision。人请求的指令就是该任务的指派指令，
                // 角色是 executor；gate 没有实现指派（与初始规划同一约定）。有了这条指派，
                // 该任务在后续返工里才有可继承的角色与可派发的入口。
                assignments: [{ taskId, role: 'executor', instruction }],
                obligations: [{
                    obligationId: 'coding-result', title: '代码满足用户要求且独立检查通过', requirementLevel: 'required', taskIds: [taskId, 'gate-goal'],
                    verificationRequirements: [
                        { requirementId: 'independent-check', requirementLevel: 'required', kind: 'dynamic', description: '候选代码需独立验证；模型完成声明不直接验收' },
                        { requirementId: 'independent-review', requirementLevel: 'required', kind: 'reviewer', description: '由独立 Reviewer 核对当前版本的需求语义、实现与工具报告' },
                    ],
                }],
                taskHierarchy: { parentOf: [{ parentTaskId: 'gate-goal', childTaskId: taskId }] },
                executionDag: { dependsOn: [{ taskId: 'gate-goal', dependsOnId: taskId, requires: { kind: 'gate-result', label: '当前代码独立检查与审阅证据' } }] },
            };
            accepted(await this.deps.control.applyPlan(buildApplyPlanCommand(draft, {
                ...this.identity(scope.projectId, `real-plan-${goalId}`), goalId, expectedRevision: goal.revision,
            })));
        }
        else ensure(await this.isRealTaskPlan(scope), '该目标的活动计划没有真实开发任务的必需要求（required 的 coding-task 及其独立验证要求）；请新建目标后执行真实任务，不要复用样例计划');
    }
    /** B-1: decide from canonical plan content, never from a `planId` prefix.
     * A plan can carry a real development task only when it declares a required
     * `coding-task` protected by required independent verification — either an
     * independent semantic review (new plans) or a required dynamic check
     * (plans accepted under the earlier tool-only contract, which must keep
     * their original obligations rather than be retrofitted). Identifier shape
     * is not evidence. */
    private async isRealTaskPlan(scope: Scope) {
        const plan = await this.deps.context.activePlan(scope);
        if (!plan) return false;
        if (!plan.tasks.some(task => task.taskId === 'coding-task' && task.requirementLevel === 'required')) return false;
        return plan.obligations.some(obligation => obligation.requirementLevel === 'required'
            && obligation.taskIds.includes('coding-task')
            && obligation.verificationRequirements.some(requirement => requirement.requirementLevel === 'required'
                && (requirement.kind === 'reviewer' || requirement.kind === 'dynamic')));
    }
    private identity(projectId: string, idempotencyKey: string) {
        return { projectId, actor: { kind: 'human' as const, id: 'user-1' }, idempotencyKey, commandId: randomUUID(), correlationId: randomUUID(), submittedAt: this.deps.now() };
    }
}
