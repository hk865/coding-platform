/**
 * P1-00 shared fixtures: workspace bootstrap.
 * MVP fixture: two isolated Project/Workspace scopes; both projects REUSE the
 * same local workspaceId ("ws-shared") — local IDs are project-local.
 */
import type { ProjectRef, ProjectSnapshot, WorkspaceRef, WorkspaceSnapshot } from "../ledger.js";
import type { BootstrapManifestEntry, BootstrapManifestSnapshot, ProjectBootstrappedEventV1, WorkspaceBootstrapCommand, WorkspaceBootstrapEntry, WorkspaceBootstrapFixtureV1, WorkspaceBootstrappedEventV1 } from "../bootstrap.js";
import { bootstrapFingerprint, bootstrapSourceDigest } from "../bootstrap.js";
import type { BootstrapLedgerCommitV1 } from "../ledger.js";
import { canonicalJson } from "../fingerprint.js";

export const WORKSPACE_BOOTSTRAP_FIXTURE_V1: WorkspaceBootstrapFixtureV1 = {
  schemaVersion: 1,
  entries: [
    { projectId: "proj-alpha", workspaceId: "ws-shared" },
    { projectId: "proj-beta", workspaceId: "ws-shared" },
  ],
};

export function fixtureDigest(): string {
  return bootstrapSourceDigest(WORKSPACE_BOOTSTRAP_FIXTURE_V1);
}

export type BootstrapEventDeps = {
  /** one eventId per emitted event (entries + unique projects) */
  eventIds: string[];
  occurredAt: string;
};

/**
 * Deterministic bootstrap event sequence for a command:
 * iterate entries in order; per unique project emit ProjectBootstrapped (at
 * first occurrence), then per entry emit WorkspaceBootstrapped.
 */
export function bootstrapEventsFor(command: WorkspaceBootstrapCommand, deps: BootstrapEventDeps): {
  events: (ProjectBootstrappedEventV1 | WorkspaceBootstrappedEventV1)[];
  eventIds: string[];
} {
  const entries = command.payload.entries;
  const uniqueProjects = new Map<string, number>();
  entries.forEach((e, index) => {
    if (!uniqueProjects.has(e.projectId)) uniqueProjects.set(e.projectId, index);
  });
  const eventCount = uniqueProjects.size + entries.length;
  if (deps.eventIds.length !== eventCount) {
    throw new Error("bootstrap eventIds length mismatch: expected " + eventCount + ", got " + deps.eventIds.length);
  }
  const events: (ProjectBootstrappedEventV1 | WorkspaceBootstrappedEventV1)[] = [];
  let n = 0;
  const idFor = (offset: number) => deps.eventIds[offset] as string;
  for (const [projectId] of uniqueProjects) {
    const eventId = idFor(n);
    n += 1;
    events.push({
      eventId,
      eventType: "ProjectBootstrapped",
      schemaVersion: 1,
      projectId,
      aggregateType: "Project",
      aggregateId: projectId,
      aggregateRevision: 1,
      causationId: command.commandId,
      correlationId: command.correlationId,
      idempotencyKey: command.identity.idempotencyKey,
      actor: { ...command.identity.actor },
      occurredAt: deps.occurredAt,
      payload: { sourceDigest: command.payload.sourceDigest },
    });
  }
  for (const entry of entries) {
    const eventId = idFor(n);
    n += 1;
    events.push({
      eventId,
      eventType: "WorkspaceBootstrapped",
      schemaVersion: 1,
      projectId: entry.projectId,
      workspaceId: entry.workspaceId,
      aggregateType: "Workspace",
      aggregateId: entry.workspaceId,
      aggregateRevision: 1,
      causationId: command.commandId,
      correlationId: command.correlationId,
      idempotencyKey: command.identity.idempotencyKey,
      actor: { ...command.identity.actor },
      occurredAt: deps.occurredAt,
      payload: { sourceDigest: command.payload.sourceDigest },
    });
  }
  return { events, eventIds: events.map((e) => e.eventId) };
}

export function bootstrapSnapshotsFor(command: WorkspaceBootstrapCommand): {
  snapshots: (ProjectSnapshot | WorkspaceSnapshot | BootstrapManifestSnapshot)[];
  manifestId: string;
} {
  const snapshots: (ProjectSnapshot | WorkspaceSnapshot | BootstrapManifestSnapshot)[] = [];
  const seenProjects = new Set<string>();
  const seenWorkspaces = new Set<string>();
  for (const entry of command.payload.entries) {
    const projectRef: ProjectRef = { aggregateType: "Project", projectId: entry.projectId };
    if (!seenProjects.has(entry.projectId)) {
      seenProjects.add(entry.projectId);
      snapshots.push({ ref: projectRef, revision: 1 });
    }
    const workspaceRef: WorkspaceRef = {
      aggregateType: "Workspace",
      projectId: entry.projectId,
      workspaceId: entry.workspaceId,
    };
    const wKey = canonicalJson(workspaceRef);
    if (!seenWorkspaces.has(wKey)) {
      seenWorkspaces.add(wKey);
      snapshots.push({ ref: workspaceRef, revision: 1 });
    }
  }
  const entries: BootstrapManifestEntry[] = command.payload.entries.map((e: WorkspaceBootstrapEntry) => ({
    projectId: e.projectId,
    workspaceId: e.workspaceId,
    projectRevision: 1 as const,
    workspaceRevision: 1 as const,
  }));
  const manifestSnapshot: BootstrapManifestSnapshot = {
    ref: { aggregateType: "BootstrapManifest", manifestId: command.payload.sourceDigest },
    revision: 1,
    schemaVersion: 1,
    sourceDigest: command.payload.sourceDigest,
    bootstrapRevision: 1,
    entries,
  };
  snapshots.push(manifestSnapshot);
  return { snapshots, manifestId: command.payload.sourceDigest };
}

export function buildBootstrapLedgerCommit(
  command: WorkspaceBootstrapCommand,
  deps: BootstrapEventDeps,
): BootstrapLedgerCommitV1 {
  const { events } = bootstrapEventsFor(command, deps);
  const { snapshots } = bootstrapSnapshotsFor(command);
  return {
    commitKind: "bootstrap",
    schemaVersion: 1,
    identity: { ...command.identity },
    fingerprint: bootstrapFingerprint(command),
    expectedVersions: [],
    events,
    snapshots,
    outboxIntents: [],
  };
}
