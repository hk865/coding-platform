/**
 * P1-02 shared fixtures: versioned governance (CompletionPolicy /
 * ArchitectureBaseline) + install/activate builders.
 *
 * Frozen semantics (see IMPLEMENTATION-HANDOFF.md "P1-02 契约与存储语义"):
 *  - fixtures have explicit schemaVersion/identity/revision/content and a
 *    canonical content digest (JCS + SHA-256 over the whole fixture);
 *  - a missing/invalid fixture is rejected at install; NO built-in content;
 *  - install persists an immutable revision (same identity/revision cannot be
 *    overwritten); install NEVER auto-activates;
 *  - activation only accepts an installed exact target ref + digest, CAS on
 *    the expected Project revision (and the per-kind active aggregate);
 *  - project active refs are independent per kind; no ArchitectureEvolutionPolicy.
 */

export type BuildGovernanceCommandDeps = {
  commandId: string;
  correlationId: string;
  submittedAt: string;
  projectId: string;

  idempotencyKey?: string;
};

export function buildInstallCommand(
  fixture: VersionedCompletionPolicyFixture | VersionedArchitectureBaselineFixture,
  deps: BuildGovernanceCommandDeps,
): GovernanceInstallCommand {
  return formalbuildInstallCommand(fixture, { ...deps, actor: deps.actor ?? { kind: "human", id: "user-1" }, idempotencyKey: deps.idempotencyKey ?? "p1-02-governance-install" });
}

/** Deterministic install ledger commit (fold-equality target for Control). */
export function buildInstallLedgerCommit(
  command: GovernanceInstallCommand,
  deps: GovernanceEventDeps,
): GovernanceInstallLedgerCommitV1 {
  if (command.commandType === "InstallCompletionPolicyRevision") {
    return {
      commitKind: "governance-install",
      schemaVersion: 1,
      identity: { ...command.identity },
      fingerprint: governanceInstallFingerprint(command),
      expectedVersions: [{ ref: completionPolicyRevisionRefFor(command), revision: 0 }],
      events: [completionPolicyInstalledEventFor(command, deps)],
      snapshots: [completionPolicyRevisionSnapshotFor(command)],
      outboxIntents: [],
    };
  }
  return {
    commitKind: "governance-install",
    schemaVersion: 1,
    identity: { ...command.identity },
    fingerprint: governanceInstallFingerprint(command),
    expectedVersions: [{ ref: architectureBaselineRevisionRefFor(command), revision: 0 }],
    events: [architectureBaselineInstalledEventFor(command, deps)],
    snapshots: [architectureBaselineRevisionSnapshotFor(command)],
    outboxIntents: [],
  };
}

export function buildActivateLedgerCommit(
  command: GovernanceActivateCommand,
  deps: GovernanceEventDeps & { activeAggregateRevision: number; projectRevision: number },
): GovernanceActivateLedgerCommitV1 {
  const projectRef = { aggregateType: "Project" as const, projectId: command.identity.projectId };
  if (command.commandType === "ActivateProjectCompletionPolicy") {
    const target = command.payload.target;
    const activeRef = {
      aggregateType: "ProjectCompletionPolicyActive" as const,
      projectId: command.identity.projectId,
    };
    const snapshot: ProjectCompletionPolicyActiveSnapshot = {
      ref: activeRef,
      projectId: command.identity.projectId,
      activeRevision: target.ref,
      revision: deps.activeAggregateRevision,
    };
    return {
      commitKind: "governance-activate",
      schemaVersion: 1,
      identity: { ...command.identity },
      fingerprint: governanceActivateFingerprint(command),
      expectedVersions: [
        { ref: projectRef, revision: deps.projectRevision },
        { ref: activeRef, revision: deps.activeAggregateRevision - 1 },
      ],
      events: [completionPolicyActivatedEventFor(command, deps)],
      snapshots: [snapshot],
      outboxIntents: [],
    };
  }
  const target = command.payload.target;
  const activeRef = {
    aggregateType: "ProjectArchitectureBaselineActive" as const,
    projectId: command.identity.projectId,
  };
  const snapshot: ProjectArchitectureBaselineActiveSnapshot = {
    ref: activeRef,
    projectId: command.identity.projectId,
    activeRevision: target.ref,
    revision: deps.activeAggregateRevision,
  };
  return {
    commitKind: "governance-activate",
    schemaVersion: 1,
    identity: { ...command.identity },
    fingerprint: governanceActivateFingerprint(command),
    expectedVersions: [
      { ref: projectRef, revision: deps.projectRevision },
      { ref: activeRef, revision: deps.activeAggregateRevision - 1 },
    ],
    events: [architectureBaselineActivatedEventFor(command, deps)],
    snapshots: [snapshot],
    outboxIntents: [],
  };
}
