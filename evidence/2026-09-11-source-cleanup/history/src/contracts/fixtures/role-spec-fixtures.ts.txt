const ROLE_SPEC_BY_ID = new Map(ROLE_SPEC_SOURCES_V1.map((source) => [source.roleId, source]));

export function roleSpecSourceFor(roleId: string): VersionedRoleSpecSourceV1 {
  const source = ROLE_SPEC_BY_ID.get(roleId);
  if (source === undefined) throw new Error("未知角色规格 source：" + roleId);
  return source;
}

/** 角色矩阵 pin：roleId + 该 source 的 revision 与内容摘要。 */
export function roleSpecPinFor(projectId: string, roleId: string): RoleSpecPinV1 {
  const source = roleSpecSourceFor(roleId);
  return { ref: roleSpecRevisionRefFor(projectId, roleId, source.revision), digest: roleSpecContentDigest(source.content, roleId, source.revision) };
}

/** 安装某个角色规格的正式命令（薄包装：字段级构造仍在契约命令层）。 */
export function buildRoleSpecInstallCommandFor(roleId: string, deps: BuildGovernanceCommandDeps): import("../role-spec.js").InstallRoleSpecRevisionCommand {
  const source = roleSpecSourceFor(roleId);
  return buildRoleSpecInstallCommand({ roleId: source.roleId, content: source.content }, deps);
}

/** 激活某个角色规格的正式命令。 */
export function buildRoleSpecActivateCommandFor(roleId: string, deps: BuildGovernanceCommandDeps & { expectedRevision: number }): import("../role-spec.js").ActivateRoleSpecRevisionCommand {
  const pin = roleSpecPinFor(deps.projectId, roleId);
  return buildRoleSpecActivateCommand(pin, deps);
}

export const RW11_ROLE_MATRIX_COORDINATOR = ROLE_SOURCE_PLANNER;

/**
 * 项目角色矩阵：默认登记全部八个角色，收敛责任默认在规划者（ARCHITECTURE：各工作包可有规划／
 * 集成负责人，跨包议题有指定收敛责任）。若调用方只登记了部分角色、且规划者不在其中，
 * 收敛责任取登记集合里的第一个角色——fixture 不制造悬空 coordinator（那会被安装守卫拒绝）。
 */
export function buildRoleMatrixFixture(projectId: string, roleIds: readonly string[] = ROLE_SPEC_SOURCES_V1.map((source) => source.roleId), coordinator: string = RW11_ROLE_MATRIX_COORDINATOR): CoordinationRoleMatrixV1 {
  const catalog: Record<string, RoleSpecPinV1> = {};
  for (const roleId of roleIds) catalog[roleId] = roleSpecPinFor(projectId, roleId);
  const resolved = roleIds.includes(coordinator) ? coordinator : roleIds[0];
  if (resolved === undefined) throw new Error("角色矩阵必须至少登记一个角色");
  return { catalog, coordinator: { roleId: resolved, note: "跨工作包议题由该角色收敛并上报需要人决定的事项" } };
}

/**
 * 带角色矩阵的协调策略正文 source：额度等其他字段与 P1-15 的既有正文同值，
 * 额外的只有 roles——安装期仍走同一条 InstallCoordinationPolicy 路径。
 */
export function buildCoordinationPolicyContentWithRolesV1(projectId: string, roleIds?: readonly string[]): CoordinationPolicyContentV1 {
  return {
    schemaVersion: 1,
    budget: { maxAutonomousReworks: P15_COORDINATION_BUDGET_MAX, maxClarifications: 8 },
    allowed: { inScopeRework: true, inScopeTesting: true },
    scope: { changesRequireHumanDecision: ["requirement", "acceptance", "baseline"] },
    upgrade: { path: "manual-decision", note: "策略变化必须由人经 install + activate 提交新 revision" },
    roles: buildRoleMatrixFixture(projectId, roleIds),
  };
}

  return {

/** 该正文装进策略时的摘要口径（与 P1-15 完全同一个函数，不另立一份）。 */
export function roleMatrixPolicyDigest(content: CoordinationPolicyContentV1, policyId: string): string {
  return coordinationPolicyContentDigest(content, policyId, P15_COORDINATION_POLICY_REVISION);
}
